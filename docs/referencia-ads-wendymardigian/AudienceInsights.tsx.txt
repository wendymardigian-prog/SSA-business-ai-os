import { useState } from "react";
import {
  BarChart, Bar,
  LineChart, Line,
  PieChart, Pie, Cell,
  XAxis, YAxis,
  CartesianGrid, Tooltip, Legend,
  ResponsiveContainer,
} from "recharts";

// ── Helpers ────────────────────────────────────────────────────────────────
export function parseAudienceData(data: any[]) {
  const getLeads = (actions: any[]) =>
    Number(actions?.find((a: any) => a.action_type === "lead")?.value ?? 0);

  return (data ?? []).map((row: any) => ({
    age:    row.age    ?? "?",
    gender: row.gender ?? "unknown",
    reach:       Number(row.reach ?? 0),
    impressions: Number(row.impressions ?? 0),
    clicks:      Number(row.clicks ?? 0),
    spend:       Number(row.spend ?? 0),
    ctr:         Number(row.ctr ?? 0),
    leads:       getLeads(row.actions ?? []),
  }));
}

export function parseHourlyData(data: any[]) {
  const getLeads = (actions: any[]) =>
    Number(actions?.find((a: any) => a.action_type === "lead")?.value ?? 0);

  const map: Record<number, any> = {};
  for (const row of data ?? []) {
    const hour = Number(row.hourly_stats_aggregated_by_advertiser_time_zone ?? row.hour ?? 0);
    map[hour] = {
      hour,
      label:       `${String(hour).padStart(2, "0")}:00`,
      spend:       Number(row.spend ?? 0),
      impressions: Number(row.impressions ?? 0),
      clicks:      Number(row.clicks ?? 0),
      ctr:         Number(row.ctr ?? 0),
      leads:       getLeads(row.actions ?? []),
    };
  }
  return Array.from({ length: 24 }, (_, h) => map[h] ?? {
    hour: h, label: `${String(h).padStart(2, "0")}:00`,
    spend: 0, impressions: 0, clicks: 0, ctr: 0, leads: 0,
  });
}

const GENDER_LABEL: Record<string, string> = {
  male: "Hombre",
  female: "Mujer",
  unknown: "Otro",
};

const GENDER_COLOR: Record<string, string> = {
  male: "#6d28d9",
  female: "#ec4899",
  unknown: "#94a3b8",
};

// ── AudienceInsights ───────────────────────────────────────────────────────
const AGE_ORDER = ["13-17", "18-24", "25-34", "35-44", "45-54", "55-64", "65+"];

export function AudienceInsights({
  data,
  compact = false,
}: {
  data: ReturnType<typeof parseAudienceData>;
  compact?: boolean;
}) {
  const [stackMetric, setStackMetric] = useState<"reach" | "leads">("reach");

  // Stacked bar data
  const stackData = AGE_ORDER
    .map((age) => {
      const rows = data.filter((r) => r.age === age);
      if (rows.length === 0) return null;
      const entry: Record<string, any> = { age };
      for (const row of rows) {
        entry[row.gender] = row[stackMetric];
      }
      return entry;
    })
    .filter(Boolean) as Record<string, any>[];

  // Donut data
  const genderTotals = ["male", "female", "unknown"]
    .map((g) => ({
      key: g,
      name: GENDER_LABEL[g],
      value: data.filter((r) => r.gender === g).reduce((s, r) => s + r.reach, 0),
      color: GENDER_COLOR[g],
    }))
    .filter((g) => g.value > 0);

  const totalReach = genderTotals.reduce((s, g) => s + g.value, 0);

  // CTR line data
  const genders = ["male", "female", "unknown"].filter((g) =>
    data.some((r) => r.gender === g && r.ctr > 0)
  );

  const ctrData = AGE_ORDER
    .map((age) => {
      const entry: Record<string, any> = { age };
      let hasData = false;
      for (const g of genders) {
        const row = data.find((r) => r.age === age && r.gender === g);
        if (row) {
          entry[g] = Number(row.ctr.toFixed(2));
          hasData = true;
        }
      }
      return hasData ? entry : null;
    })
    .filter(Boolean) as Record<string, any>[];

  const fmtK = (v: number) =>
    v >= 1000 ? `${(v / 1000).toFixed(1)}k` : String(Math.round(v));

  const stackedGenders = ["male", "female", "unknown"].filter((g) =>
    stackData.some((d) => d[g] != null && d[g] > 0)
  );

  return (
    <div className="grid grid-cols-1 md:grid-cols-4 gap-6">
      {/* Columna izquierda — 2 gráficos por edad apilados (75%) */}
      <div className="md:col-span-3 space-y-6">
        {/* Sección A — Stacked bar */}
        <div>
          <div className="flex items-center justify-between mb-3">
            <p className="text-xs text-muted-foreground font-medium uppercase tracking-wider">
              {stackMetric === "reach" ? "Alcance" : "Leads"} por edad y género
            </p>
            <div className="flex gap-1">
              {(["reach", "leads"] as const).map((m) => (
                <button
                  key={m}
                  onClick={() => setStackMetric(m)}
                  className={`px-2.5 py-0.5 rounded text-xs transition-colors ${
                    stackMetric === m
                      ? "bg-primary text-primary-foreground"
                      : "bg-muted text-muted-foreground hover:text-foreground"
                  }`}
                >
                  {m === "reach" ? "Alcance" : "Leads"}
                </button>
              ))}
            </div>
          </div>

          <ResponsiveContainer width="100%" height={compact ? 150 : 200}>
            <BarChart data={stackData} margin={{ top: 4, right: 4, left: 0, bottom: 0 }}>
              <CartesianGrid strokeDasharray="3 3" stroke="hsl(var(--border))" vertical={false} />
              <XAxis
                dataKey="age"
                tick={{ fontSize: 11, fill: "hsl(var(--muted-foreground))" }}
                axisLine={false}
                tickLine={false}
              />
              <YAxis
                tick={{ fontSize: 11, fill: "hsl(var(--muted-foreground))" }}
                axisLine={false}
                tickLine={false}
                tickFormatter={fmtK}
                width={36}
              />
              <Tooltip
                contentStyle={{
                  backgroundColor: "hsl(var(--card))",
                  border: "1px solid hsl(var(--border))",
                  borderRadius: "8px",
                  fontSize: "12px",
                }}
                formatter={(value: number, name: string) => [
                  fmtK(value),
                  GENDER_LABEL[name] ?? name,
                ]}
              />
              <Legend
                formatter={(value) => GENDER_LABEL[value as string] ?? value}
                wrapperStyle={{ fontSize: "12px" }}
              />
              {stackedGenders.map((g, i, arr) => (
                <Bar
                  key={g}
                  dataKey={g}
                  stackId="audience"
                  fill={GENDER_COLOR[g]}
                  opacity={0.85}
                  radius={i === arr.length - 1 ? [3, 3, 0, 0] : [0, 0, 0, 0]}
                  maxBarSize={48}
                />
              ))}
            </BarChart>
          </ResponsiveContainer>
        </div>

        {/* Sección C — CTR lines (oculto en compact) */}
        {!compact && (
          <div>
            <p className="text-xs text-muted-foreground font-medium uppercase tracking-wider mb-3">
              CTR por franja de edad
            </p>
            {ctrData.length > 0 ? (
              <ResponsiveContainer width="100%" height={160}>
                <LineChart data={ctrData} margin={{ top: 4, right: 4, left: 0, bottom: 0 }}>
                  <CartesianGrid strokeDasharray="3 3" stroke="hsl(var(--border))" vertical={false} />
                  <XAxis
                    dataKey="age"
                    tick={{ fontSize: 10, fill: "hsl(var(--muted-foreground))" }}
                    axisLine={false}
                    tickLine={false}
                  />
                  <YAxis
                    tick={{ fontSize: 10, fill: "hsl(var(--muted-foreground))" }}
                    axisLine={false}
                    tickLine={false}
                    tickFormatter={(v) => `${v}%`}
                    width={32}
                  />
                  <Tooltip
                    contentStyle={{
                      backgroundColor: "hsl(var(--card))",
                      border: "1px solid hsl(var(--border))",
                      borderRadius: "8px",
                      fontSize: "12px",
                    }}
                    formatter={(v: number, name: string) => [
                      `${v.toFixed(2)}%`,
                      GENDER_LABEL[name] ?? name,
                    ]}
                  />
                  {genders.map((g) => (
                    <Line
                      key={g}
                      type="monotone"
                      dataKey={g}
                      stroke={GENDER_COLOR[g]}
                      strokeWidth={2}
                      dot={{ r: 3, fill: GENDER_COLOR[g] }}
                      activeDot={{ r: 4 }}
                    />
                  ))}
                </LineChart>
              </ResponsiveContainer>
            ) : (
              <p className="text-xs text-muted-foreground">Sin datos de CTR disponibles</p>
            )}
          </div>
        )}
      </div>

      {/* Columna derecha — Donut (25%, alto completo, leyenda debajo) */}
      <div className="md:col-span-1 flex flex-col">
        <p className="text-xs text-muted-foreground font-medium uppercase tracking-wider mb-3">
          Distribución por género
        </p>
        <div className="flex-1 flex flex-col items-center justify-center gap-4">
          <ResponsiveContainer width="100%" height={180}>
            <PieChart>
              <Pie
                data={genderTotals}
                cx="50%"
                cy="50%"
                innerRadius={50}
                outerRadius={80}
                dataKey="value"
                strokeWidth={2}
                stroke="hsl(var(--background))"
              >
                {genderTotals.map((entry, index) => (
                  <Cell key={`cell-${index}`} fill={entry.color} />
                ))}
              </Pie>
              <Tooltip
                contentStyle={{
                  backgroundColor: "hsl(var(--card))",
                  border: "1px solid hsl(var(--border))",
                  borderRadius: "8px",
                  fontSize: "12px",
                }}
                formatter={(value: number) => [fmtK(value), "Alcance"]}
              />
            </PieChart>
          </ResponsiveContainer>

          <div className="space-y-2 w-full">
            {genderTotals.map((g) => (
              <div key={g.key} className="flex items-center gap-2">
                <span
                  className="w-2.5 h-2.5 rounded-full shrink-0"
                  style={{ backgroundColor: g.color }}
                />
                <span className="text-xs text-muted-foreground">{g.name}</span>
                <span className="text-xs font-medium ml-auto text-foreground">
                  {totalReach > 0 ? ((g.value / totalReach) * 100).toFixed(0) : 0}%
                </span>
              </div>
            ))}
          </div>
        </div>
      </div>
    </div>
  );
}


// ── HourlyChart ────────────────────────────────────────────────────────────
export function HourlyChart({
  data,
  metric,
}: {
  data: ReturnType<typeof parseHourlyData>;
  metric: "spend" | "ctr" | "clicks";
}) {
  const fmtTick = (v: number) =>
    metric === "spend" ? `$${v.toFixed(0)}` :
    metric === "ctr"   ? `${v.toFixed(1)}%` :
    String(Math.round(v));

  const fmtTooltip = (v: number) =>
    metric === "spend" ? `$${v.toFixed(2)}` :
    metric === "ctr"   ? `${v.toFixed(2)}%` :
    String(Math.round(v));

  const metricLabel =
    metric === "spend" ? "Gasto" :
    metric === "ctr"   ? "CTR" : "Clics";

  const maxVal = Math.max(...data.map((d) => d[metric] as number), 1);

  return (
    <ResponsiveContainer width="100%" height={180}>
      <BarChart data={data} margin={{ top: 4, right: 4, left: 0, bottom: 0 }}>
        <CartesianGrid strokeDasharray="3 3" stroke="hsl(var(--border))" vertical={false} />
        <XAxis
          dataKey="label"
          tick={{ fontSize: 10, fill: "hsl(var(--muted-foreground))" }}
          interval={2}
          axisLine={false}
          tickLine={false}
        />
        <YAxis
          tick={{ fontSize: 10, fill: "hsl(var(--muted-foreground))" }}
          axisLine={false}
          tickLine={false}
          tickFormatter={fmtTick}
          width={40}
        />
        <Tooltip
          contentStyle={{
            backgroundColor: "hsl(var(--card))",
            border: "1px solid hsl(var(--border))",
            borderRadius: "8px",
            fontSize: "12px",
          }}
          formatter={(v: number) => [fmtTooltip(v), metricLabel]}
          labelFormatter={(label) => `Hora: ${label}`}
        />
        <Bar dataKey={metric} radius={[2, 2, 0, 0]} maxBarSize={24}>
          {data.map((entry, index) => {
            const intensity = maxVal > 0 ? (entry[metric] as number) / maxVal : 0;
            const opacity = 0.25 + intensity * 0.75;
            return <Cell key={`cell-${index}`} fill={`rgba(124, 58, 237, ${opacity})`} />;
          })}
        </Bar>
      </BarChart>
    </ResponsiveContainer>
  );
}
