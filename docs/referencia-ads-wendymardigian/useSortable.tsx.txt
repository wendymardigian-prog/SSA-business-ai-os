import { useState, useMemo } from "react";
import { ArrowUp, ArrowDown, ArrowUpDown } from "lucide-react";

export type SortDir = "asc" | "desc";

export function useSortable<T extends Record<string, any>>(rows: T[], initialKey: keyof T | null = null, initialDir: SortDir = "desc") {
  const [sortKey, setSortKey] = useState<keyof T | null>(initialKey);
  const [sortDir, setSortDir] = useState<SortDir>(initialDir);

  const handleSort = (key: keyof T) => {
    if (sortKey === key) {
      setSortDir(sortDir === "asc" ? "desc" : "asc");
    } else {
      setSortKey(key);
      setSortDir("desc");
    }
  };

  const sorted = useMemo(() => {
    if (!sortKey) return rows;
    const copy = [...rows];
    copy.sort((a, b) => {
      const av = a[sortKey];
      const bv = b[sortKey];
      const an = typeof av === "number" ? av : Number(av);
      const bn = typeof bv === "number" ? bv : Number(bv);
      let cmp: number;
      if (!isNaN(an) && !isNaN(bn) && typeof av !== "string" && typeof bv !== "string") {
        cmp = an - bn;
      } else {
        cmp = String(av ?? "").localeCompare(String(bv ?? ""));
      }
      return sortDir === "asc" ? cmp : -cmp;
    });
    return copy;
  }, [rows, sortKey, sortDir]);

  return { sorted, sortKey, sortDir, handleSort };
}

interface SortableThProps<T> {
  label: string;
  sortKey: keyof T;
  activeKey: keyof T | null;
  dir: SortDir;
  onSort: (key: keyof T) => void;
  right?: boolean;
  className?: string;
}

export function SortableTh<T>({
  label, sortKey, activeKey, dir, onSort, right = true, className = "",
}: SortableThProps<T>) {
  const isActive = activeKey === sortKey;
  const Icon = !isActive ? ArrowUpDown : dir === "asc" ? ArrowUp : ArrowDown;
  return (
    <th
      onClick={() => onSort(sortKey)}
      className={`py-2 px-2 text-xs uppercase tracking-wider font-medium cursor-pointer select-none transition-colors ${
        isActive ? "text-foreground" : "text-muted-foreground hover:text-foreground"
      } ${right ? "text-right" : "text-left"} ${className}`}
    >
      <span className={`inline-flex items-center gap-1 ${right ? "justify-end w-full" : ""}`}>
        {label}
        <Icon className={`w-3 h-3 ${isActive ? "opacity-100" : "opacity-40"}`} />
      </span>
    </th>
  );
}
