function TipTh({
  label, align, tipMap,
}: { label: string; align: "left" | "right"; tipMap: Record<string, { title: string; body: string }> }) {
  const tip = tipMap[label];
  const base = `py-2 px-2 text-xs uppercase tracking-wider font-medium text-muted-foreground ${align === "right" ? "text-right" : "text-left"}`;
  if (!tip) return <th className={base}>{label}</th>;
  return (
    <th className={base}>
      <UITooltip>
        <UITooltipTrigger asChild>
          <span className="cursor-help underline decoration-dotted decoration-muted-foreground/40 underline-offset-4">{label}</span>
        </UITooltipTrigger>
        <UITooltipContent side="top" align="start" className="max-w-xs text-xs leading-relaxed normal-case tracking-normal text-left">
          <p className="font-semibold text-foreground mb-1 text-left">{tip.title}</p>
          <p className="text-muted-foreground text-left">{tip.body}</p>
        </UITooltipContent>
      </UITooltip>
    </th>
  );
}

// ── Helpers ───────────────────────────────────────────────────────────────────
const fmt = (n: string | number | undefined, decimals = 0) =>
  Number(n ?? 0).toLocaleString("es-AR", { minimumFractionDigits: decimals, maximumFractionDigits: decimals });

const fmtMoney = (n: string | number | undefined) => `$${fmt(n, 2)}`;

const getLeads = (actions?: Array<{ action_type: string; value: string }>) => {
  if (!actions) return 0;
  return actions
    .filter(a => a.action_type.includes("lead") || a.action_type.includes("complete_registration"))
    .reduce((acc, a) => acc + Number(a.value), 0);
};

const getActionValue = (actions: Array<{ action_type: string; value: string }> | undefined, type: string) => {
  if (!actions) return 0;
  return actions.filter(a => a.action_type === type).reduce((acc, a) => acc + Number(a.value), 0);
};

const getVideoActionValue = (actions: Array<{ action_type: string; value: string }> | undefined) => {
  if (!actions || actions.length === 0) return 0;
  return Number(actions[0].value ?? 0);
};

async function callAds(action: string, params = {}) {
  const { data, error } = await supabase.functions.invoke("meta-ads", {
    body: { action, ...params },
  });
  if (error) throw error;
  return data;
}

// ── Pantalla de conexión ──────────────────────────────────────────────────────
function ConnectAds({ onConnected, expiredMessage }: { onConnected: () => void; expiredMessage?: string }) {
  const [token, setToken]         = useState("");
  const [accountId, setAccountId] = useState("");
  const [loading, setLoading]     = useState(false);
  const [error, setError]         = useState("");

  const handleConnect = async () => {
    if (!token.trim() || !accountId.trim()) return;
    setLoading(true);
    setError("");
    try {
      const data = await callAds("connect", {
        access_token: token.trim(),
        ad_account_id: accountId.trim(),
      });
      if (!data?.ok) {
        setError(data?.error ?? "Error al conectar. Verificá el token y el Account ID.");
        return;
      }
      onConnected();
    } catch {
      setError("Error inesperado. Intentá de nuevo.");
    } finally {
      setLoading(false);
    }
  };


  return (
    <div className="max-w-xl mx-auto space-y-6 p-6 rounded-2xl border border-border bg-card">
      <div className="flex items-center justify-center w-14 h-14 rounded-xl bg-primary/10 mx-auto">
        <BarChart2 className="w-7 h-7 text-primary" />
      </div>
      <div className="text-center space-y-2">
        <h2 className="text-2xl font-bold text-foreground">
          {expiredMessage ? "Reconectar Meta Ads" : "Conectar Meta Ads"}
        </h2>
        <p className="text-sm text-muted-foreground">
          Ingresá tu token y Ad Account ID para ver el rendimiento de tus campañas.
        </p>
      </div>
      {expiredMessage && (
        <div className="flex items-start gap-2 text-sm bg-amber-50 border border-amber-200 text-amber-900 rounded-lg p-3">
          <AlertCircle className="w-4 h-4 mt-0.5 shrink-0" />
          <div>
            <p className="font-medium">Token expirado</p>
            <p className="text-xs mt-0.5">{expiredMessage}</p>
          </div>
        </div>
      )}
      <div className="space-y-2 text-sm text-muted-foreground bg-muted/40 rounded-lg p-4">
        <p className="font-medium text-foreground">Cómo obtener los datos:</p>
        <ul className="list-disc list-inside space-y-1">
          <li>Generá un token en Meta for Developers con permiso <code>ads_read</code></li>
          <li>Tu Ad Account ID lo encontrás en Meta Business Suite → Configuración → Cuentas publicitarias (formato: <code>1234567890</code>)</li>
          <li><strong>Tip:</strong> para evitar reconexiones cada ~60 días, usá un token de <em>System User</em> en Business Manager → Configuración → Usuarios del sistema.</li>
        </ul>
      </div>
      <div className="space-y-3">
        <Input
          placeholder="Access Token (EAA...)"
          value={token}
          onChange={(e) => setToken(e.target.value)}
          className="font-mono text-xs"
        />
        <Input
          placeholder="Ad Account ID (1234567890)"
          value={accountId}
          onChange={(e) => setAccountId(e.target.value)}
        />
        {error && (
          <div className="flex items-center gap-2 text-sm text-destructive">
            <AlertCircle className="w-4 h-4" /> {error}
          </div>
        )}
        <Button onClick={handleConnect} disabled={loading} className="w-full" variant="hero">
          {loading ? <Loader2 className="w-4 h-4 animate-spin" /> : null}
          Conectar
        </Button>
      </div>
    </div>
  );
}

// ── KPI Card pequeño ──────────────────────────────────────────────────────────
function KpiCard({ label, value, sub, valueClass }: {
  label: string; value: string; sub?: string; valueClass?: string;
}) {
  // detect delta-style sub (starts with ▲ / ▼)
  const isUp = sub?.trim().startsWith("▲");
  const isDown = sub?.trim().startsWith("▼");
  return (
    <div className="admin-kpi">
      <p className="admin-kpi-label">{label}</p>
      <p className={`admin-kpi-value ${valueClass ?? ""}`}>{value}</p>
      {sub && (
        <p className="admin-kpi-sub">
          {isUp && <span className="admin-delta-up">{sub}</span>}
          {isDown && <span className="admin-delta-down">{sub}</span>}
          {!isUp && !isDown && <span>{sub}</span>}
        </p>
      )}
    </div>
  );
}

// ── Status dot ────────────────────────────────────────────────────────────────
function StatusDot({ status }: { status?: string }) {
  const isActive = status === "ACTIVE" || !status;
  return (
    <span className="inline-flex items-center gap-1.5">
      <span className={`w-1.5 h-1.5 rounded-full ${isActive ? "bg-green-500" : "bg-muted-foreground"}`} />
      <span className={isActive ? "text-green-600" : "text-muted-foreground"}>
        {isActive ? "Activa" : "Pausada"}
      </span>
    </span>
  );
}

// ── Ranking badge ─────────────────────────────────────────────────────────────
function RankingBadge({ value }: { value?: string }) {
  if (!value || value === "UNKNOWN") return <span className="text-muted-foreground">—</span>;
  if (value === "ABOVE_AVERAGE") return <span className="text-[10px] px-1.5 py-0.5 rounded bg-green-100 text-green-800">▲ Superior</span>;
  if (value === "AVERAGE") return <span className="text-[10px] px-1.5 py-0.5 rounded bg-yellow-100 text-yellow-800">— Promedio</span>;
  return <span className="text-[10px] px-1.5 py-0.5 rounded bg-red-100 text-red-800">▼ Inferior</span>;
}

                        <td className="py-2 px-2 text-right text-foreground">{fmt(totalLeads)}</td>
                      </tr>
                    </tfoot>
                  </table>
                </div>
              )}
            </TabsContent>

            {/* ADSETS */}
            <TabsContent value="adsets">
              {loadingSecondary && adsets.length === 0 ? (
                <div className="flex items-center justify-center py-12"><Loader2 className="w-5 h-5 animate-spin text-primary" /></div>
              ) : adsets.length === 0 ? (
                <div className="flex flex-col items-center justify-center py-10 text-xs text-muted-foreground gap-2">
                  <BarChart2 className="w-8 h-8 opacity-40" /> Sin ad sets en este período
                </div>
              ) : (
                <div className="overflow-x-auto">
                  <table className="w-full text-sm">
                    <thead>
                      <tr className="border-b border-border">
                        {["Ad Set","Estado","Gasto","Impr.","Alcance","Frec.","Clics","CTR","CPM","CPC","Leads","CPL"].map((h, i) => (
                          <TipTh key={h} label={h} align={i >= 2 ? "right" : "left"} tipMap={COL_TIPS} />
                        ))}
                      </tr>
                    </thead>
                    <tbody>
                      {adsets.map((a: any, i) => {
                        const aImpr = Number(a.impressions ?? 0);
                        const aReach = Number(a.reach ?? 0);
                        const aClicks = Number(a.clicks ?? 0);
                        const aSpend = Number(a.spend ?? 0);
                        const aCtr = Number(a.ctr ?? 0);
                        const aFreq = Number(a.frequency ?? (aReach > 0 ? aImpr / aReach : 0));
                        const aCpm = Number(a.cpm ?? (aImpr > 0 ? (aSpend / aImpr) * 1000 : 0));
                        const aCpc = aClicks > 0 ? aSpend / aClicks : 0;
                        const aLeads = getLeads(a.actions);
                        const aCpl = aLeads > 0 ? aSpend / aLeads : 0;
                        const ctrClass = aCtr > 3.5 ? "text-green-600" : aCtr < 2 ? "text-destructive" : "text-foreground";
                        return (
                          <tr key={i} className="border-b border-border/50 hover:bg-muted/30">
                            <td className="py-2 px-2 max-w-[200px] truncate font-medium text-foreground" title={a.adset_name}>{a.adset_name}</td>
                            <td className="py-2 px-2 text-xs"><StatusDot status={a.status} /></td>
                            <td className="py-2 px-2 text-right text-foreground">{fmtMoney(aSpend)}</td>
                            <td className="py-2 px-2 text-right text-foreground">{fmt(aImpr)}</td>
                            <td className="py-2 px-2 text-right text-foreground">{fmt(aReach)}</td>
                            <td className="py-2 px-2 text-right text-foreground">{aFreq > 0 ? aFreq.toFixed(2) : "—"}</td>
                            <td className="py-2 px-2 text-right text-foreground">{fmt(aClicks)}</td>
                            <td className={`py-2 px-2 text-right ${ctrClass}`}>{aCtr.toFixed(2)}%</td>
                            <td className="py-2 px-2 text-right text-foreground">{fmtMoney(aCpm)}</td>
                            <td className="py-2 px-2 text-right text-foreground">{fmtMoney(aCpc)}</td>
                            <td className={`py-2 px-2 text-right ${aLeads === 0 ? "text-destructive font-bold" : "text-foreground font-semibold"}`}>{fmt(aLeads)}</td>
                            <td className="py-2 px-2 text-right text-foreground">{aLeads > 0 ? fmtMoney(aCpl) : "—"}</td>
                          </tr>
                        );
                      })}
                    </tbody>
                  </table>
                </div>
              )}
            </TabsContent>

            {/* ANUNCIOS */}
            <TabsContent value="ads">
              {loadingSecondary && adsData.length === 0 ? (
                <div className="flex items-center justify-center py-12"><Loader2 className="w-5 h-5 animate-spin text-primary" /></div>
              ) : adsData.length === 0 ? (
                <div className="flex flex-col items-center justify-center py-10 text-xs text-muted-foreground gap-2">
                  <BarChart2 className="w-8 h-8 opacity-40" /> Sin anuncios en este período
                </div>
              ) : (
                <div className="overflow-x-auto">
                  <table className="w-full text-sm">
                    <thead>
                      <tr className="border-b border-border">
                        {["Anuncio","Gasto","Impr.","Clics","CTR","CPC","Leads","Calidad","Eng.","Conv."].map((h, i) => (
                          <TipTh key={h} label={h} align={i === 0 ? "left" : "right"} tipMap={COL_TIPS_ADS} />
                        ))}
                      </tr>
                    </thead>
                    <tbody>
                      {adsData.map((a: any, i) => {
                        const aSpend = Number(a.spend ?? 0);
                        const aImpr = Number(a.impressions ?? 0);
                        const aClicks = Number(a.clicks ?? 0);
                        const aCtr = Number(a.ctr ?? 0);
                        const aCpc = Number(a.cpc ?? (aClicks > 0 ? aSpend / aClicks : 0));
                        const aLeads = getLeads(a.actions);
                        const ctrClass = aCtr > 3.5 ? "text-green-600" : aCtr < 2 ? "text-destructive" : "text-foreground";
                        return (
                          <tr key={i} className="border-b border-border/50 hover:bg-muted/30 cursor-pointer" onClick={() => openAdDrawer(a)}>
                            <td className="py-2 px-2 max-w-[200px] truncate font-medium text-foreground" title={a.ad_name}>{a.ad_name}</td>
                            <td className="py-2 px-2 text-right text-foreground">{fmtMoney(aSpend)}</td>
                            <td className="py-2 px-2 text-right text-foreground">{fmt(aImpr)}</td>
                            <td className="py-2 px-2 text-right text-foreground">{fmt(aClicks)}</td>
                            <td className={`py-2 px-2 text-right ${ctrClass}`}>{aCtr.toFixed(2)}%</td>
                            <td className="py-2 px-2 text-right text-foreground">{fmtMoney(aCpc)}</td>
                            <td className={`py-2 px-2 text-right ${aLeads === 0 ? "text-destructive font-bold" : "text-foreground font-semibold"}`}>{fmt(aLeads)}</td>
                            <td className="py-2 px-2 text-right"><RankingBadge value={a.quality_ranking} /></td>
                            <td className="py-2 px-2 text-right"><RankingBadge value={a.engagement_rate_ranking} /></td>
                            <td className="py-2 px-2 text-right"><RankingBadge value={a.conversion_rate_ranking} /></td>
                          </tr>
                        );
                      })}
                    </tbody>
                  </table>
                </div>
              )}
            </TabsContent>
          </Tabs>
        </div>
      </div>

      {/* SECCIÓN — MÉTRICAS POR DÍA (account-level) */}
      {loadingDaily && accountDailyData.length === 0 ? (
        <div className="rounded-xl border border-border bg-card p-3 space-y-3">
          <h3 className="text-sm font-semibold text-foreground flex items-center gap-2">
            <CalendarDays className="w-4 h-4 text-primary" /> Métricas por día
          </h3>
          <SectionSkeleton rows={6} height="h-6" />
        </div>
      ) : accountDailyData.length > 0 && (
        <div className="rounded-xl border border-border bg-card p-3 space-y-3">
          <h3 className="text-sm font-semibold text-foreground flex items-center gap-2">
            <CalendarDays className="w-4 h-4 text-primary" /> Métricas por día
          </h3>
          <div className="overflow-x-auto">
            <table className="w-full text-sm">
              <thead>
                <tr className="border-b border-border">
                  <SortableTh<AccountDailyRow> label="Fecha"     sortKey="date"        activeKey={accDailySortKey} dir={accDailySortDir} onSort={accDailySort} right={false} />
                  <SortableTh<AccountDailyRow> label="Gasto"     sortKey="spend"       activeKey={accDailySortKey} dir={accDailySortDir} onSort={accDailySort} />
                  <SortableTh<AccountDailyRow> label="Impr."     sortKey="impressions" activeKey={accDailySortKey} dir={accDailySortDir} onSort={accDailySort} />
                  <SortableTh<AccountDailyRow> label="Alcance"   sortKey="reach"       activeKey={accDailySortKey} dir={accDailySortDir} onSort={accDailySort} />
                  <SortableTh<AccountDailyRow> label="Frec."     sortKey="frequency"   activeKey={accDailySortKey} dir={accDailySortDir} onSort={accDailySort} />
                  <SortableTh<AccountDailyRow> label="Clics"     sortKey="clicks"      activeKey={accDailySortKey} dir={accDailySortDir} onSort={accDailySort} />
                  <SortableTh<AccountDailyRow> label="CTR"       sortKey="ctr"         activeKey={accDailySortKey} dir={accDailySortDir} onSort={accDailySort} />
                  <SortableTh<AccountDailyRow> label="CPM"       sortKey="cpm"         activeKey={accDailySortKey} dir={accDailySortDir} onSort={accDailySort} />
                  <SortableTh<AccountDailyRow> label="CPC"       sortKey="cpc"         activeKey={accDailySortKey} dir={accDailySortDir} onSort={accDailySort} />
                  <SortableTh<AccountDailyRow> label="Leads"     sortKey="leads"       activeKey={accDailySortKey} dir={accDailySortDir} onSort={accDailySort} />
                </tr>
              </thead>
              <tbody>
                {sortedAccountDaily.map((d, i) => (
                  <tr key={i} className="border-b border-border/50 hover:bg-muted/30">
                    <td className="py-2 px-2 text-foreground">{d.date}</td>
                    <td className="py-2 px-2 text-right text-foreground">${d.spend.toFixed(2)}</td>
                    <td className="py-2 px-2 text-right text-foreground">{d.impressions.toLocaleString()}</td>
                    <td className="py-2 px-2 text-right text-foreground">{d.reach.toLocaleString()}</td>
                    <td className="py-2 px-2 text-right text-foreground">{d.frequency > 0 ? d.frequency.toFixed(2) : "—"}</td>
                    <td className="py-2 px-2 text-right text-foreground">{d.clicks.toLocaleString()}</td>
                    <td className="py-2 px-2 text-right text-foreground">{d.ctr.toFixed(2)}%</td>
                    <td className="py-2 px-2 text-right text-foreground">${d.cpm.toFixed(2)}</td>
                    <td className="py-2 px-2 text-right text-foreground">${d.cpc.toFixed(2)}</td>
                    <td className={`py-2 px-2 text-right ${d.leads === 0 ? "text-muted-foreground" : "text-foreground font-semibold"}`}>{d.leads}</td>
                  </tr>
                ))}
              </tbody>
              <tfoot>
                <tr className="border-t border-border font-semibold">
                  <td className="py-2 px-2 text-foreground">Total</td>
                  <td className="py-2 px-2 text-right text-foreground">${accDailyTotals.spend.toFixed(2)}</td>
                  <td className="py-2 px-2 text-right text-foreground">{accDailyTotals.impressions.toLocaleString()}</td>
                  <td className="py-2 px-2 text-right text-foreground">{accDailyTotals.reach.toLocaleString()}</td>
                  <td className="py-2 px-2 text-right text-muted-foreground">—</td>
                  <td className="py-2 px-2 text-right text-foreground">{accDailyTotals.clicks.toLocaleString()}</td>
                  <td className="py-2 px-2 text-right text-muted-foreground">—</td>
                  <td className="py-2 px-2 text-right text-muted-foreground">—</td>
                  <td className="py-2 px-2 text-right text-muted-foreground">—</td>
                  <td className="py-2 px-2 text-right text-foreground">{accDailyTotals.leads}</td>
                </tr>
              </tfoot>
            </table>
          </div>
        </div>
      )}
      </div>

      {/* Drawers */}
      {selectedCampaign && (
        <CampaignDrawer
          campaign={selectedCampaign}
          dailyData={campaignDailyData}
          loading={drawerLoading}
          onClose={() => { setSelectedCampaign(null); setCampaignDailyData([]); }}
          since={since}
          until={until}
        />
      )}
      {selectedAd && (
        <AdDrawer
          ad={selectedAd}
          dailyData={adDailyData}
          loading={drawerLoading}
          onClose={() => { setSelectedAd(null); setAdDailyData([]); }}
          since={since}
          until={until}
        />
      )}

      {/* Panel de análisis IA — lazy mount: solo se monta la primera vez que se abre */}
      {aiPanelEverOpened && (
        <AdsAIChatPanel
          open={aiPanelOpen}
          onOpenChange={setAiPanelOpen}
          snapshot={loadingPrimary ? null : adsSnapshot}
        />
      )}
    </div>
    </UITooltipProvider>
  );
}

// ── Campaign Drawer ──────────────────────────────────────────────────────────
function CampaignDrawer({
  campaign, dailyData, loading, onClose, since, until,
}: {
  campaign: any; dailyData: any[]; loading: boolean; onClose: () => void; since: string; until: string;
}) {
  const spend = Number(campaign.spend ?? 0);
  const impr = Number(campaign.impressions ?? 0);
  const reach = Number(campaign.reach ?? 0);
  const clicks = Number(campaign.clicks ?? 0);
  const ctr = Number(campaign.ctr ?? 0);
  const leads = getLeads(campaign.actions);
  const cpl = leads > 0 ? spend / leads : 0;
  const freq = reach > 0 ? impr / reach : 0;
  const cpm = impr > 0 ? (spend / impr) * 1000 : 0;
  const cpc = clicks > 0 ? spend / clicks : 0;

  const [metric, setMetric] = useState<"gasto"|"clics"|"ctr"|"impresiones">("gasto");

  const chartData = [...dailyData]
    .sort((a, b) => (a.date_start ?? "").localeCompare(b.date_start ?? ""))
    .map(d => ({
      date: d.date_start?.slice(5) ?? "",
      gasto: Number(d.spend ?? 0),
      clics: Number(d.clicks ?? 0),
      ctr: Number(d.ctr ?? 0),
      impresiones: Number(d.impressions ?? 0),
    }));

  return (
    <Sheet open={true} onOpenChange={(open) => { if (!open) onClose(); }}>
      <SheetContent side="right" className="w-[90vw] max-w-[800px] overflow-y-auto">
        <div className="space-y-5">
          <div className="space-y-1">
            <p className="text-[10px] uppercase tracking-wider text-muted-foreground font-medium">Campaña</p>
            <h2 className="text-lg font-bold text-foreground">{campaign.campaign_name}</h2>
            <p className="text-xs text-muted-foreground">{since} → {until}</p>
          </div>

          <div className="grid grid-cols-4 gap-2">
            <KpiCard label="Gasto" value={fmtMoney(spend)} valueClass="text-primary" />
            <KpiCard label="Impresiones" value={fmt(impr)} />
            <KpiCard label="Alcance" value={fmt(reach)} />
            <KpiCard label="Clics" value={fmt(clicks)} sub={`CTR: ${ctr.toFixed(2)}%`} />
            <KpiCard label="Leads" value={fmt(leads)} sub={leads > 0 ? `CPL: ${fmtMoney(cpl)}` : "CPL: —"} valueClass={leads === 0 ? "text-destructive" : "text-foreground"} />
            <KpiCard label="Frec." value={freq > 0 ? freq.toFixed(2) : "—"} />
            <KpiCard label="CPM" value={fmtMoney(cpm)} />
            <KpiCard label="CPC" value={fmtMoney(cpc)} />
          </div>

          <div className="rounded-xl border border-border bg-card p-3 space-y-3">
            <div className="flex items-center justify-between flex-wrap gap-2">
              <h3 className="text-sm font-semibold text-foreground">Evolución diaria</h3>
              <div className="flex items-center gap-1 bg-muted/30 rounded-md p-0.5">
                <MiniToggle active={metric === "gasto"} onClick={() => setMetric("gasto")}>Gasto</MiniToggle>
                <MiniToggle active={metric === "impresiones"} onClick={() => setMetric("impresiones")}>Impr.</MiniToggle>
                <MiniToggle active={metric === "clics"} onClick={() => setMetric("clics")}>Clics</MiniToggle>
                <MiniToggle active={metric === "ctr"} onClick={() => setMetric("ctr")}>CTR</MiniToggle>
              </div>
            </div>
            <div className="h-[220px]">
              {loading ? (
                <div className="h-full flex items-center justify-center"><Loader2 className="w-5 h-5 animate-spin text-primary" /></div>
              ) : chartData.length === 0 ? (
                <div className="h-full flex items-center justify-center text-xs text-muted-foreground">Sin datos diarios</div>
              ) : (
                <ResponsiveContainer width="100%" height="100%">
                  <LineChart data={chartData} margin={{ top: 5, right: 8, left: -20, bottom: 0 }}>
                    <CartesianGrid strokeDasharray="3 3" stroke="hsl(var(--border))" opacity={0.3} />
                    <XAxis dataKey="date" stroke="hsl(var(--muted-foreground))" fontSize={10} tickLine={false} axisLine={false} />
                    <YAxis stroke="hsl(var(--muted-foreground))" fontSize={10} tickLine={false} axisLine={false} />
                    <Tooltip
                      contentStyle={{ background: "hsl(var(--card))", border: "1px solid hsl(var(--border))", borderRadius: 8, fontSize: 12 }}
                      formatter={(v: number) => [
                        metric === "gasto" ? fmtMoney(v) : metric === "ctr" ? `${v.toFixed(2)}%` : fmt(v),
                        metric.charAt(0).toUpperCase() + metric.slice(1),
                      ]}
                    />
                    <Line type="monotone" dataKey={metric} stroke="hsl(var(--primary))" strokeWidth={2} dot={false} />
                  </LineChart>
                </ResponsiveContainer>
              )}
            </div>
          </div>
        </div>
      </SheetContent>
    </Sheet>
  );
}

// ── Ad Drawer ────────────────────────────────────────────────────────────────
function AdDrawer({
  ad, dailyData, loading, onClose, since, until,
}: {
  ad: any; dailyData: any[]; loading: boolean; onClose: () => void; since: string; until: string;
}) {
  const spend = Number(ad.spend ?? 0);
  const impr = Number(ad.impressions ?? 0);
  const reach = Number(ad.reach ?? 0);
  const clicks = Number(ad.clicks ?? 0);
  const ctr = Number(ad.ctr ?? 0);
  const cpc = Number(ad.cpc ?? (clicks > 0 ? spend / clicks : 0));
  const leads = getLeads(ad.actions);
  const cpl = leads > 0 ? spend / leads : 0;
  const freq = Number(ad.frequency ?? 0);
  const cpm = impr > 0 ? (spend / impr) * 1000 : 0;

  const [metric, setMetric] = useState<"gasto"|"clics"|"ctr"|"impresiones">("gasto");
  const [adPlacements, setAdPlacements] = useState<any[]>([]);
  const [loadingPlacement, setLoadingPlacement] = useState(false);

  useEffect(() => {
    if (!ad.ad_id) return;
    setLoadingPlacement(true);
    callAds("ad_placement", { ad_id: ad.ad_id, since, until })
      .then(res => setAdPlacements((res as any)?.data ?? []))
      .catch(() => {})
      .finally(() => setLoadingPlacement(false));
  }, [ad.ad_id, since, until]);

  const chartData = [...dailyData]
    .sort((a, b) => (a.date_start ?? "").localeCompare(b.date_start ?? ""))
    .map(d => ({
      date: d.date_start?.slice(5) ?? "",
      gasto: Number(d.spend ?? 0),
      clics: Number(d.clicks ?? 0),
      ctr: Number(d.ctr ?? 0),
      impresiones: Number(d.impressions ?? 0),
    }));

  const rankings = [
    { label: "Calidad del anuncio", value: ad.quality_ranking },
    { label: "Tasa de engagement", value: ad.engagement_rate_ranking },
    { label: "Tasa de conversión", value: ad.conversion_rate_ranking },
  ];

  return (
    <Sheet open={true} onOpenChange={(open) => { if (!open) onClose(); }}>
      <SheetContent side="right" className="w-[90vw] max-w-[800px] overflow-y-auto">
        <div className="space-y-5">
          <div className="space-y-1">
            <p className="text-[10px] uppercase tracking-wider text-muted-foreground font-medium">Anuncio</p>
            <h2 className="text-lg font-bold text-foreground">{ad.ad_name}</h2>
            <p className="text-xs text-muted-foreground">{since} → {until}</p>
          </div>

          <div className="grid grid-cols-4 gap-2">
            <KpiCard label="Gasto" value={fmtMoney(spend)} valueClass="text-primary" />
            <KpiCard label="Impresiones" value={fmt(impr)} />
            <KpiCard label="Alcance" value={fmt(reach)} />
            <KpiCard label="Clics" value={fmt(clicks)} sub={`CTR: ${ctr.toFixed(2)}%`} />
            <KpiCard label="Leads" value={fmt(leads)} sub={leads > 0 ? `CPL: ${fmtMoney(cpl)}` : "CPL: —"} valueClass={leads === 0 ? "text-destructive" : "text-foreground"} />
            <KpiCard label="CPC" value={fmtMoney(cpc)} />
            <KpiCard label="CPM" value={fmtMoney(cpm)} />
            <KpiCard label="Frec." value={freq > 0 ? freq.toFixed(2) : "—"} />
          </div>

          <div className="rounded-xl border border-border bg-card p-3 space-y-3">
            <h3 className="text-sm font-semibold text-foreground">Rankings de calidad</h3>
            {rankings.map((r, i) => {
              const score = r.value === "ABOVE_AVERAGE" ? 85 : r.value === "AVERAGE" ? 50 : r.value === "BELOW_AVERAGE" ? 20 : 0;
              const color = r.value === "ABOVE_AVERAGE" ? "bg-green-500" : r.value === "AVERAGE" ? "bg-yellow-400" : r.value === "BELOW_AVERAGE" ? "bg-red-500" : "bg-muted";
              const label = r.value === "ABOVE_AVERAGE" ? "▲ Superior" : r.value === "AVERAGE" ? "— Promedio" : r.value === "BELOW_AVERAGE" ? "▼ Inferior" : "Sin datos";
              return (
                <div key={i} className="space-y-1">
                  <div className="flex items-center justify-between text-xs">
                    <span className="text-muted-foreground">{r.label}</span>
                    <span className="text-foreground font-medium">{label}</span>
                  </div>
                  <div className="h-2 bg-muted rounded overflow-hidden">
                    <div className={`h-full ${color} rounded`} style={{ width: `${score}%` }} />
                  </div>
                </div>
              );
            })}
          </div>

          {/* Placement del ad */}
          <PlacementCard placements={adPlacements} loading={loadingPlacement} />

          <div className="rounded-xl border border-border bg-card p-3 space-y-3">
            <div className="flex items-center justify-between flex-wrap gap-2">
              <h3 className="text-sm font-semibold text-foreground">Evolución diaria</h3>
              <div className="flex items-center gap-1 bg-muted/30 rounded-md p-0.5">
                <MiniToggle active={metric === "gasto"} onClick={() => setMetric("gasto")}>Gasto</MiniToggle>
                <MiniToggle active={metric === "impresiones"} onClick={() => setMetric("impresiones")}>Impr.</MiniToggle>
                <MiniToggle active={metric === "clics"} onClick={() => setMetric("clics")}>Clics</MiniToggle>
                <MiniToggle active={metric === "ctr"} onClick={() => setMetric("ctr")}>CTR</MiniToggle>
              </div>
            </div>
            <div className="h-[220px]">
              {loading ? (
                <div className="h-full flex items-center justify-center"><Loader2 className="w-5 h-5 animate-spin text-primary" /></div>
              ) : chartData.length === 0 ? (
                <div className="h-full flex items-center justify-center text-xs text-muted-foreground">Sin datos diarios</div>
              ) : (
                <ResponsiveContainer width="100%" height="100%">
                  <LineChart data={chartData} margin={{ top: 5, right: 8, left: -20, bottom: 0 }}>
                    <CartesianGrid strokeDasharray="3 3" stroke="hsl(var(--border))" opacity={0.3} />
                    <XAxis dataKey="date" stroke="hsl(var(--muted-foreground))" fontSize={10} tickLine={false} axisLine={false} />
                    <YAxis stroke="hsl(var(--muted-foreground))" fontSize={10} tickLine={false} axisLine={false} />
                    <Tooltip
                      contentStyle={{ background: "hsl(var(--card))", border: "1px solid hsl(var(--border))", borderRadius: 8, fontSize: 12 }}
                      formatter={(v: number) => [
                        metric === "gasto" ? fmtMoney(v) : metric === "ctr" ? `${v.toFixed(2)}%` : fmt(v),
                        metric.charAt(0).toUpperCase() + metric.slice(1),
                      ]}
                    />
                    <Line type="monotone" dataKey={metric} stroke="hsl(var(--primary))" strokeWidth={2} dot={false} />
                  </LineChart>
                </ResponsiveContainer>
