// ── Helpers ─────────────────────────────────────────────────────────────────
const fmt = (n: string | number | undefined, decimals = 0) =>
  Number(n ?? 0).toLocaleString("es-AR", {
    minimumFractionDigits: decimals,
    maximumFractionDigits: decimals,
  });

const fmtMoney = (n: string | number | undefined) => `$${fmt(n, 2)}`;

const getLeads = (actions?: Array<{ action_type: string; value: string }>) => {
  if (!actions) return 0;
  return actions
    .filter(a => a.action_type.includes("lead") || a.action_type.includes("complete_registration"))
    .reduce((acc, a) => acc + Number(a.value), 0);
};

const getAction = (actions: any[] | undefined, type: string): number => {
  if (!actions) return 0;
  return actions
    .filter(a => a.action_type === type)
    .reduce((acc, a) => acc + Number(a.value), 0);
};

const getVideoValue = (arr: any[] | undefined): number => {
  if (!arr || arr.length === 0) return 0;
  return Number(arr[0].value ?? 0);
};

const getOutboundClicks = (data: any): number => {
  if (!data) return 0;
  if (Array.isArray(data.outbound_clicks))
    return data.outbound_clicks.reduce((acc: number, a: any) => acc + Number(a.value ?? 0), 0);
  return 0;
};

async function callAds(action: string, params: Record<string, any> = {}) {
  const { data, error } = await supabase.functions.invoke("meta-ads", {
    body: { action, ...params },
  });
  if (error) throw error;
  return data;
}

// ── Toggle ──────────────────────────────────────────────────────────────────
function MiniToggle({ active, onClick, children }: {
  active: boolean; onClick: () => void; children: React.ReactNode;
}) {
  return (
    <button
      onClick={onClick}
      className={`text-xs px-2.5 py-1 rounded-md transition-colors ${
        active ? "bg-primary text-primary-foreground" : "text-muted-foreground hover:bg-muted/50"
      }`}
    >
      {children}
    </button>
  );
}

// ── Ranking bar ─────────────────────────────────────────────────────────────
function RankingBar({ label, value }: { label: string; value?: string }) {
  const score = value === "ABOVE_AVERAGE" ? 85 : value === "AVERAGE" ? 50 : value === "BELOW_AVERAGE" ? 20 : 0;
  const color = value === "ABOVE_AVERAGE" ? "bg-green-500" : value === "AVERAGE" ? "bg-yellow-400" : value === "BELOW_AVERAGE" ? "bg-red-500" : "bg-muted";
  const text  = value === "ABOVE_AVERAGE" ? "▲ Superior" : value === "AVERAGE" ? "— Promedio" : value === "BELOW_AVERAGE" ? "▼ Inferior" : "Sin datos";
  const tCls  = value === "ABOVE_AVERAGE" ? "text-green-600 font-medium"
              : value === "BELOW_AVERAGE" ? "text-destructive font-medium"
              : "text-muted-foreground";
  return (
    <div className="space-y-1.5">
      <div className="flex items-center justify-between text-xs">
        <span className="text-muted-foreground">{label}</span>
        <span className={tCls}>{text}</span>
      </div>
      <div className="h-1.5 bg-muted rounded-full overflow-hidden">
        <div className={`h-full ${color} rounded-full transition-all`} style={{ width: `${score}%` }} />
      </div>
    </div>
  );
}

    switch (period) {
      case "last_7":  { const d = new Date(today); d.setDate(d.getDate()-7);  return { since: d.toISOString().split("T")[0], until }; }
      case "last_30": { const d = new Date(today); d.setDate(d.getDate()-30); return { since: d.toISOString().split("T")[0], until }; }

      default: {

        return { since, until };

    }

  const loadData = useCallback(async () => {

    // Cache hit
    const cached = cachedDataRef.current;

      setAdMeta(cached.adMeta ?? null);
      setDailyData(cached.dailyData ?? []);

      setLoadingMeta(false);

          placements: freshPlacements,
          actionsData: freshActions,
          videoData: freshVideo,
          funnelData: freshFunnel,
          hourlyData: freshHourly,

          setLoadingAudience(false);
          setRefreshing(false);

            ...snapshotRef.current,
            audienceData: freshAudience,
          });
        })

          setLoadingAudience(false);
          setRefreshing(false);

      setLoadingChart(false);

  const freq           = Number(ad.frequency ?? (reach > 0 ? impr / reach : 0));

          variant="outline"
          size="sm"
          className="h-8 rounded-lg bg-white border-[hsl(var(--admin-border-strong))] text-[hsl(var(--admin-strong))] hover:bg-[hsl(var(--admin-soft))]"
          onClick={loadData}
          disabled={refreshing}
        >
          <RefreshCw className={`w-3.5 h-3.5 ${refreshing ? "animate-spin" : ""}`} />
          <span className="hidden sm:inline">Actualizar</span>
        </Button>
      </header>

      <div className="p-6 space-y-3">
        {/* Título + creative type pill */}
        <div className="flex flex-wrap items-center gap-3">
          <h2 className="text-xl font-bold text-foreground leading-tight">
            {ad.ad_name}
          </h2>
          {creativeType && (
            <span className="text-[10px] px-2 py-0.5 rounded-full border border-primary/30 bg-primary/10 text-primary font-medium">
              {CREATIVE_TYPE_LABELS[creativeType] ?? creativeType}
            </span>
          )}
        </div>

        {/* SECCIÓN 1 — KPI ROW (single card) */}
        <div className="space-y-2">
          <p className="admin-section-label">Resumen</p>
          <div className="bg-card border border-border rounded-xl p-3">
            <div className="grid grid-cols-4 lg:grid-cols-8 gap-3 divide-x divide-border/60">
              {[
                { key: "spend",       label: "Gasto total", value: fmtMoney(spend),                              sub: "presupuesto usado",                                                          trend: trends.spend },
                { key: "impressions", label: "Impresiones", value: fmt(impr),                                    sub: "total de vistas",                                                            trend: trends.impressions },
                { key: "reach",       label: "Alcance",     value: fmt(reach),                                   sub: "cuentas únicas",                                                             trend: trends.reach },
                { key: "frequency",   label: "Frecuencia",  value: freq > 0 ? freq.toFixed(2) : "—",             sub: "imp / persona",                                                              trend: null },
                { key: "clicks",      label: "Clics",       value: fmt(clicks),                                  sub: outboundClicks > 0 ? `Salientes: ${fmt(outboundClicks)} (${outboundCtr.toFixed(2)}%)` : `CTR: ${ctr.toFixed(2)}%`, trend: trends.clicks },
                { key: "cpm",         label: "CPM",         value: fmtMoney(cpm),                                sub: "por mil impr.",                                                              trend: trends.cpm },
                { key: "cpc",         label: "CPC",         value: fmtMoney(cpc),                                sub: "por clic",                                                                   trend: trends.cpc },
                { key: "leads",       label: "Leads",       value: fmt(leads),                                   sub: leads > 0 ? `CPL: ${fmtMoney(cpl)}` : "CPL: —",                               trend: trends.leads },
              ].map(({ key, label, value, sub, trend }, idx) => (
                <div key={key} className={`px-3 ${idx === 0 ? "pl-0" : ""}`}>
                  <p className="text-[10px] uppercase tracking-wider text-muted-foreground font-medium leading-none mb-1.5">{label}</p>
                  <p className="text-xl font-bold leading-none text-foreground">{value}</p>
                  <div className="flex items-center justify-between mt-1.5 gap-1">
                    <p className="text-[10px] text-muted-foreground truncate">{sub}</p>
                    {trend != null && (
                      <span className={`text-[10px] font-medium shrink-0 ${trend >= 0 ? "text-emerald-500" : "text-red-500"}`}>
                        {trend >= 0 ? "▲" : "▼"}{Math.abs(trend).toFixed(1)}%
                      </span>
                    )}
                  </div>
                </div>
              ))}
            </div>
          </div>
        </div>

        {/* SECCIÓN 2 — EVOLUCIÓN DIARIA */}
        <div className="admin-card p-3 space-y-3">
          <div className="flex items-center justify-between flex-wrap gap-3">
            <h3 className="text-sm font-semibold text-foreground flex items-center gap-2">
              <Activity className="w-4 h-4 text-primary" /> Evolución diaria
            </h3>
            <div className="flex items-center gap-3 text-xs flex-wrap">
              <div className="flex items-center gap-1.5">
                <span className="inline-block w-3 h-3 rounded-sm" style={{ backgroundColor: METRIC_CONFIG[leftMetric]?.color }} />
                <span className="text-muted-foreground">Eje izq:</span>
                <select value={leftMetric} onChange={(e) => setLeftMetric(e.target.value)}
                  className="bg-muted border border-border rounded px-1.5 py-0.5 text-xs text-foreground cursor-pointer">
                  {LEFT_OPTIONS.map((key) => <option key={key} value={key}>{METRIC_CONFIG[key].label}</option>)}
                </select>
              </div>
              <div className="flex items-center gap-1.5">
                <span className="inline-block w-5 h-0.5 rounded" style={{ backgroundColor: METRIC_CONFIG[rightMetric]?.color }} />
                <span className="text-muted-foreground">Eje der:</span>
                <select value={rightMetric} onChange={(e) => setRightMetric(e.target.value)}
                  className="bg-muted border border-border rounded px-1.5 py-0.5 text-xs text-foreground cursor-pointer">
                  {RIGHT_OPTIONS.map((key) => <option key={key} value={key}>{METRIC_CONFIG[key].label}</option>)}
                </select>
              </div>
            </div>
          </div>
          <div className="h-72">
            {loading && chartData.length === 0 ? (
              <div className="h-full flex items-center justify-center"><Loader2 className="w-5 h-5 animate-spin text-primary" /></div>
            ) : chartData.length === 0 ? (
              <div className="h-full flex items-center justify-center text-sm text-muted-foreground">
                Sin datos para el período seleccionado
              </div>
            ) : (
              <ResponsiveContainer width="100%" height="100%">
                <ComposedChart data={chartData} margin={{ top: 4, right: 16, left: 0, bottom: 0 }}>
                  <CartesianGrid strokeDasharray="3 3" stroke="#ece9fe" vertical={false} />
                  <XAxis dataKey="date" tick={{ fontSize: 11, fill: "#6b6b8a" }} axisLine={false} tickLine={false} />
                  <YAxis yAxisId="left" orientation="left" tick={{ fontSize: 11, fill: "#6b6b8a" }} axisLine={false} tickLine={false}
                    tickFormatter={(v) => METRIC_CONFIG[leftMetric]?.format(v) ?? v} width={52} />
                  <YAxis yAxisId="right" orientation="right" tick={{ fontSize: 11, fill: "#6b6b8a" }} axisLine={false} tickLine={false}
                    tickFormatter={(v) => METRIC_CONFIG[rightMetric]?.format(v) ?? v} width={44} />
                  <RechartsTooltip
                    contentStyle={{ background: "#fff", border: "1px solid #ece9fe", borderRadius: 12, fontSize: 12, boxShadow: "0 8px 24px rgba(124,58,237,0.12)" }}
                    formatter={(value: number, name: string) => {
                      const cfg = METRIC_CONFIG[name];
                      return [cfg ? cfg.format(value) : value, cfg?.label ?? name];
                    }}
                  />
                  <Legend formatter={(value) => METRIC_CONFIG[value as string]?.label ?? value} wrapperStyle={{ fontSize: 12, paddingTop: 8 }} />
                  <Bar yAxisId="left" dataKey={leftMetric} fill="#7c3aed" radius={[6, 6, 0, 0]} maxBarSize={40} />
                  <Line yAxisId="right" type="monotone" dataKey={rightMetric} stroke="#ec4899" strokeWidth={2}
                    dot={{ r: 3, fill: "#ec4899" }} activeDot={{ r: 5 }} />
                </ComposedChart>
              </ResponsiveContainer>
            )}
          </div>
        </div>

        {/* SECCIÓN — Acciones | Placement | Rankings | Video */}
        <div className="grid grid-cols-1 sm:grid-cols-2 lg:grid-cols-4 gap-3">
          {/* Acciones generadas */}
          {(() => {
            const ACTION_LABELS: Record<string, string> = {
              link_click: "Clics en enlace",
              lead: "Leads",
              video_view: "Reproducciones",
              post_engagement: "Engagement",
              post_reaction: "Reacciones",
              comment: "Comentarios",
              post: "Compartidos",
              omni_purchase: "Compras",
              omni_add_to_cart: "Add to cart",
              complete_registration: "Registros",
            };
            const SHOW = ["link_click","lead","video_view","post_engagement","post_reaction","comment","post"];
            const actions = aggregatedActions
              .filter((a: any) => SHOW.includes(a.action_type) && Number(a.value) > 0)
              .map((a: any) => ({ type: a.action_type, label: ACTION_LABELS[a.action_type] ?? a.action_type, value: Number(a.value) }))
              .sort((a, b) => b.value - a.value);
            const maxVal = Math.max(...actions.map((a) => a.value), 1);
            return (
              <div className="rounded-xl border border-border bg-card p-3 space-y-3">
                <h3 className="text-sm font-semibold text-foreground flex items-center gap-2">
                  <Zap className="w-4 h-4 text-primary" /> Acciones generadas
                </h3>
                {actions.length === 0 ? (
                  <div className="h-[140px] flex items-center justify-center text-xs text-muted-foreground">Sin datos</div>
                ) : (
                  <div className="space-y-1.5">
                    {actions.map((a) => (
                      <div key={a.type} className="flex items-center gap-2 text-xs">
                        <div className="flex-1 min-w-0 truncate text-muted-foreground" title={a.label}>{a.label}</div>
                        <div className="w-16 h-2 bg-muted rounded overflow-hidden">
                          <div className="h-full bg-primary rounded" style={{ width: `${Math.min((a.value / maxVal) * 100, 100)}%` }} />
                        </div>
                        <div className="w-12 text-right text-foreground font-medium">
                          {a.value >= 1000 ? `${(a.value / 1000).toFixed(1)}k` : a.value}
                        </div>
                      </div>
                    ))}
                  </div>
                )}
              </div>
            );
          })()}

          {/* Por placement */}
          <div className="rounded-xl border border-border bg-card p-3 space-y-3">
            <div className="flex items-center justify-between flex-wrap gap-2">
              <h3 className="text-sm font-semibold text-foreground flex items-center gap-2">
                <Target className="w-4 h-4 text-primary" /> Por placement
              </h3>
              <div className="flex items-center gap-1 bg-muted/30 rounded-md p-0.5">
                <MiniToggle active={placementToggle === "spend"} onClick={() => setPlacementToggle("spend")}>Gasto</MiniToggle>
                <MiniToggle active={placementToggle === "ctr"} onClick={() => setPlacementToggle("ctr")}>CTR</MiniToggle>
              </div>
            </div>
            {loading && placementAgg.length === 0 ? (
              <div className="h-[140px] flex items-center justify-center"><Loader2 className="w-5 h-5 animate-spin text-primary" /></div>
            ) : placementAgg.length === 0 ? (
              <div className="h-[140px] flex items-center justify-center text-xs text-muted-foreground">Sin datos</div>
            ) : (
              <>
                <div className="space-y-1.5">
                  {placementSorted.slice(0, 5).map((p, i) => {
                    const value = placementToggle === "spend" ? p.spend : p.ctr;
                    const pct = placementTotal > 0 ? (value / placementTotal) * 100 : 0;
                    return (
                      <div key={i} className="flex items-center gap-2 text-xs">
                        <div className="flex-1 min-w-0 truncate text-muted-foreground" title={p.name}>{p.name}</div>
                        <div className="w-12 h-2 bg-muted rounded overflow-hidden">
                          <div className="h-full bg-primary rounded" style={{ width: `${Math.min(pct, 100)}%` }} />
                        </div>
                        <div className="w-12 text-right text-foreground font-medium">
                          {placementToggle === "spend" ? fmtMoney(value) : `${value.toFixed(2)}%`}
                        </div>
                      </div>
                    );
                  })}
                </div>
                {bestCtrPlacement && (
                  <div className="text-[10px] pt-2 border-t border-border space-y-0.5">
                    <p className="text-green-600 truncate">▲ {bestCtrPlacement.name} {bestCtrPlacement.ctr.toFixed(2)}%</p>
                    {worstCtrPlacement && worstCtrPlacement.name !== bestCtrPlacement.name && (
                      <p className="text-destructive truncate">▼ {worstCtrPlacement.name} {worstCtrPlacement.ctr.toFixed(2)}%</p>
                    )}
                  </div>
                )}
              </>
            )}
          </div>

          {/* Rankings de calidad */}
          <div className="rounded-xl border border-border bg-card p-3 space-y-3">
            <h3 className="text-sm font-semibold text-foreground flex items-center gap-2">
              <BarChart2 className="w-4 h-4 text-primary" /> Rankings
            </h3>
            {!hasRankings ? (
              <div className="h-[140px] flex items-center justify-center text-xs text-muted-foreground">Sin datos</div>
            ) : (
              <div className="space-y-3 pt-1">
                <RankingBar label="Calidad"     value={ad.quality_ranking} />
                <RankingBar label="Engagement"  value={ad.engagement_rate_ranking} />
                <RankingBar label="Conversión"  value={ad.conversion_rate_ranking} />
              </div>
            )}
          </div>

          {/* Retención de video */}
          <div className="rounded-xl border border-border bg-card p-3 space-y-3">
            <h3 className="text-sm font-semibold text-foreground flex items-center gap-2">
              <Zap className="w-4 h-4 text-primary" /> Retención de video
            </h3>
            {loading && !videoData ? (
              <div className="h-[140px] flex items-center justify-center"><Loader2 className="w-5 h-5 animate-spin text-primary" /></div>
            ) : !videoStages ? (
              <div className="h-[140px] flex flex-col items-center justify-center text-xs text-muted-foreground gap-2">
                <VideoOff className="w-7 h-7 opacity-40" /> Sin datos de video
              </div>
            ) : (
              <>
                <div className="flex items-end justify-between gap-1 h-[95px] pt-5">
                  {videoStages.map((s, i) => (
                    <div key={i} className="flex-1 flex flex-col items-center gap-1">
                      <span className="text-[9px] font-bold text-foreground">{s.pct.toFixed(0)}%</span>
                      <div className="w-full bg-primary rounded-t" style={{ height: `${Math.max((s.pct / 100) * 75, 2)}px` }} />
                    </div>
                  ))}
                </div>
                <div className="flex justify-between gap-1">
                  {videoStages.map((s, i) => (
                    <span key={i} className="flex-1 text-center text-[9px] text-muted-foreground">{s.label}</span>
                  ))}
                </div>
                <div className="pt-2 border-t border-border flex justify-between text-[11px]">
                  <span className="text-muted-foreground">T. prom <span className="text-foreground font-semibold">{avgWatchTime.toFixed(1)}s</span></span>
                  <span className="text-muted-foreground">Thru <span className="text-foreground font-semibold">{fmt(thruPlays)}</span></span>
                </div>
              </>
            )}
          </div>
        </div>

        {/* ── Audiencia: Edad & Género ─────────────────────────────────────── */}
        {audienceData.length > 0 && (
          <div className="rounded-xl border border-border bg-card p-3 space-y-3">
            <p className="text-sm font-semibold text-foreground flex items-center gap-2">
              <Users className="w-4 h-4 text-primary" />
              Audiencia — Edad & Género
            </p>
            <AudienceInsights data={audienceData} />
          </div>
        )}

        {/* ── Desglose horario ─────────────────────────────────────────────── */}
        {hourlyData.some((h) => h.spend > 0 || h.clicks > 0) && (
          <div className="rounded-xl border border-border bg-card p-3 space-y-3">
            <div className="flex items-center justify-between flex-wrap gap-2">
              <p className="text-sm font-semibold text-foreground flex items-center gap-2">
                <Clock className="w-4 h-4 text-primary" />
                Rendimiento por hora del día
              </p>
              <div className="flex gap-1">
                {(["spend", "ctr", "clicks"] as const).map((m) => (
                  <button
                    key={m}
                    onClick={() => setHourlyMetric(m)}
                    className={`px-2.5 py-1 rounded-lg text-xs transition-colors ${
                      hourlyMetric === m
                        ? "bg-primary text-primary-foreground"
                        : "bg-muted text-muted-foreground hover:text-foreground"
                    }`}
                  >
                    {m === "spend" ? "Gasto" : m === "ctr" ? "CTR" : "Clics"}
                  </button>
                ))}
              </div>
            </div>
            <HourlyChart data={hourlyData} metric={hourlyMetric} />
          </div>
        )}

        {/* ── Funnel de conversión ─────────────────────────────────────────── */}
        {funnelData && (() => {
          const fImpr      = Number(funnelData.impressions ?? 0);
          const fClicks    = Number(funnelData.outbound_clicks?.[0]?.value ?? funnelData.clicks ?? 0);
          const fLeads     = getLeadsFromActions(funnelData.actions ?? []);
          const fPurchases = getPurchases(funnelData.actions ?? []);
          const fRoas      = getRoas(funnelData.purchase_roas ?? []);
          const fConvValue = getConversionValue(funnelData.action_values ?? []);
          if (fLeads === 0 && fPurchases === 0) return null;

          const steps = [
            { label: "Impresiones", value: fImpr,   pct: 100,
              fmt: fImpr >= 1000 ? `${(fImpr/1000).toFixed(1)}k` : String(fImpr) },
            { label: "Clics",       value: fClicks, pct: fImpr > 0 ? (fClicks / fImpr) * 100 : 0,
              fmt: String(fClicks) },
            { label: "Leads",       value: fLeads,  pct: fClicks > 0 ? (fLeads / fClicks) * 100 : 0,
              fmt: String(fLeads) },
            ...(fPurchases > 0 ? [{
              label: "Compras", value: fPurchases,
              pct: fLeads > 0 ? (fPurchases / fLeads) * 100 : (fClicks > 0 ? (fPurchases / fClicks) * 100 : 0),
              fmt: String(fPurchases),
            }] : []),
          ];

          return (
            <div className="rounded-xl border border-border bg-card p-3 space-y-3">
              <h3 className="text-sm font-semibold text-foreground flex items-center gap-2">
                <TrendingDown className="w-4 h-4 text-primary" />
                Funnel de conversión
              </h3>
              <FunnelViz steps={steps} roas={fRoas} convValue={fConvValue} />
            </div>
          );
        })()}

        {/* ── Tabla métricas por día ──────────────────────────────────────── */}
        <div className="rounded-xl border border-border bg-card p-3 space-y-3">
          <h3 className="text-sm font-semibold text-foreground flex items-center gap-2">
            <CalendarDays className="w-4 h-4 text-primary" /> Métricas por día
          </h3>
          {loading && sortedDaily.length === 0 ? (
            <div className="py-8 flex items-center justify-center">
              <Loader2 className="w-5 h-5 animate-spin text-muted-foreground" />
            </div>
          ) : sortedDaily.length === 0 ? (
            <div className="py-8 text-center text-sm text-muted-foreground">Sin datos diarios</div>
          ) : (
            <div className="overflow-x-auto">
              <table className="w-full text-sm">
                <thead>
                  <tr className="border-b border-border">
                    <SortableTh<DailyRow> label="Fecha"   sortKey="date"        activeKey={sortKey} dir={sortDir} onSort={handleSort} right={false} />
                    <SortableTh<DailyRow> label="Gasto"   sortKey="spend"       activeKey={sortKey} dir={sortDir} onSort={handleSort} />
                    <SortableTh<DailyRow> label="Impr."   sortKey="impressions" activeKey={sortKey} dir={sortDir} onSort={handleSort} />
                    <SortableTh<DailyRow> label="Alcance" sortKey="reach"       activeKey={sortKey} dir={sortDir} onSort={handleSort} />
                    <SortableTh<DailyRow> label="Clics"   sortKey="clicks"      activeKey={sortKey} dir={sortDir} onSort={handleSort} />
                    <SortableTh<DailyRow> label="CTR"     sortKey="ctr"         activeKey={sortKey} dir={sortDir} onSort={handleSort} />
                    <SortableTh<DailyRow> label="CPM"     sortKey="cpm"         activeKey={sortKey} dir={sortDir} onSort={handleSort} />
                    <SortableTh<DailyRow> label="CPC"     sortKey="cpc"         activeKey={sortKey} dir={sortDir} onSort={handleSort} />
                    <SortableTh<DailyRow> label="Leads"   sortKey="leads"       activeKey={sortKey} dir={sortDir} onSort={handleSort} />
                  </tr>
                </thead>
                <tbody>
                  {sortedDaily.map((d, i) => (
                    <tr key={i} className="border-b border-border/50 hover:bg-muted/30">
                      <td className="py-2 px-2 text-foreground">{d.date}</td>
                      <td className="py-2 px-2 text-right text-foreground">{fmtMoney(d.spend)}</td>
                      <td className="py-2 px-2 text-right text-foreground">{fmt(d.impressions)}</td>
                      <td className="py-2 px-2 text-right text-foreground">{fmt(d.reach)}</td>
                      <td className="py-2 px-2 text-right text-foreground">{fmt(d.clicks)}</td>
                      <td className={`py-2 px-2 text-right ${d.ctr > 3.5 ? "text-green-600" : d.ctr < 2 && d.ctr > 0 ? "text-destructive" : "text-foreground"}`}>
                        {d.ctr.toFixed(2)}%
                      </td>
                      <td className="py-2 px-2 text-right text-foreground">{fmtMoney(d.cpm)}</td>
                      <td className="py-2 px-2 text-right text-foreground">{fmtMoney(d.cpc)}</td>
                      <td className={`py-2 px-2 text-right ${d.leads === 0 ? "text-muted-foreground" : "text-foreground"}`}>
                        {fmt(d.leads)}
                      </td>
                    </tr>
                  ))}
                </tbody>
                <tfoot>
                  <tr className="border-t border-border font-semibold">
                    <td className="py-2 px-2 text-foreground">Total</td>
                    <td className="py-2 px-2 text-right text-foreground">{fmtMoney(dailyTotals.spend)}</td>
                    <td className="py-2 px-2 text-right text-foreground">{fmt(dailyTotals.impressions)}</td>
                    <td className="py-2 px-2 text-right text-foreground">{fmt(dailyTotals.reach)}</td>
                    <td className="py-2 px-2 text-right text-foreground">{fmt(dailyTotals.clicks)}</td>
                    <td className="py-2 px-2 text-right text-foreground">—</td>
                    <td className="py-2 px-2 text-right text-foreground">—</td>
                    <td className="py-2 px-2 text-right text-foreground">—</td>
                    <td className="py-2 px-2 text-right text-foreground">{fmt(dailyTotals.leads)}</td>
                  </tr>
                </tfoot>
              </table>
            </div>
          )}
        </div>
      </div>
    </div>
  );
}
