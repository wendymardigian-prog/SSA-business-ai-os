// ── Helpers ──────────────────────────────────────────────────────────────────
const fmt = (n: string | number | undefined, decimals = 0) =>
  Number(n ?? 0).toLocaleString("es-AR", {
    minimumFractionDigits: decimals,
    maximumFractionDigits: decimals,
  });
const fmtMoney = (n: string | number | undefined) => `$${fmt(n, 2)}`;

const getLeads = (actions?: Array<{ action_type: string; value: string }>) => {
  if (!actions) return 0;
  return actions
    .filter(a => a.action_type.includes("lead") || a.action_type.includes("complete_registration"))
    .reduce((acc, a) => acc + Number(a.value), 0);
};
const getVideoActionValue = (actions: Array<{ action_type: string; value: string }> | undefined) => {
  if (!actions || actions.length === 0) return 0;
  return Number(actions[0].value ?? 0);
};

async function callAds(action: string, params = {}) {
  const { data, error } = await supabase.functions.invoke("meta-ads", {
    body: { action, ...params },
  });
  if (error) throw error;
  return data;
}

// ── Toggle button ────────────────────────────────────────────────────────────
function MiniToggle({ active, onClick, children }: { active: boolean; onClick: () => void; children: React.ReactNode }) {
  return (
    <button
      onClick={onClick}
      className={`text-xs px-2.5 py-1 rounded-md transition-colors ${
        active ? "bg-primary text-primary-foreground" : "text-muted-foreground hover:bg-muted/50"
      }`}
    >
      {children}
    </button>
  );
}

// ── Ranking badge ────────────────────────────────────────────────────────────
function RankBadge({ value }: { value?: string }) {
  if (!value || value === "UNKNOWN") return <span className="text-muted-foreground">—</span>;
  if (value === "ABOVE_AVERAGE") return <span className="text-[10px] px-1.5 py-0.5 rounded bg-green-100 text-green-800">▲ Superior</span>;
  if (value === "AVERAGE") return <span className="text-[10px] px-1.5 py-0.5 rounded bg-yellow-100 text-yellow-800">— Promedio</span>;
  return <span className="text-[10px] px-1.5 py-0.5 rounded bg-red-100 text-red-800">▼ Inferior</span>;
}
function StatusDot({ status }: { status?: string }) {
  const isActive = status === "ACTIVE" || !status;
  return (
    <span className="inline-flex items-center gap-1.5">
      <span className={`w-1.5 h-1.5 rounded-full ${isActive ? "bg-green-500" : "bg-muted-foreground"}`} />
      <span className={isActive ? "text-green-600" : "text-muted-foreground"}>
        {isActive ? "Activa" : "Pausada"}
      </span>
    </span>
  );
}

interface Props {
  campaign: any;
  onBack: () => void;
  onOpenAd?: (ad: any) => void;
  initialAds?: any[];
  initialAdsets?: any[];
  cachedData?: {
    campaignMeta?: any;
    dailyData?: any[];
    ads?: any[];
    adsets?: any[];
    adsDailyData?: any[];
    placements?: any[];
    audienceData?: any[];
    hourlyData?: any[];
    funnelData?: any;
    devices?: any[];
    videoData?: any;
  } | null;
  onCacheData?: (data: any) => void;
}

export default function CampaignDetailPanel({
  campaign, onBack, onOpenAd,
  initialAds, initialAdsets, cachedData, onCacheData,
}: Props) {
  const [period, setPeriod] = useState("last_90");
  const [refreshing, setRefreshing] = useState(false);

  const [dailyData, setDailyData] = useState<any[]>([]);
  const [adsDailyData, setAdsDailyData] = useState<any[]>([]);
  const [adsets, setAdsets] = useState<any[]>(initialAdsets ?? []);
  const [ads, setAds] = useState<any[]>(initialAds ?? []);
  const [placements, setPlacements] = useState<any[]>([]);
  const [campaignMeta, setCampaignMeta] = useState<any | null>(null);
  const [audienceData, setAudienceData] = useState<ReturnType<typeof parseAudienceData>>([]);
  const [hourlyData, setHourlyData] = useState<ReturnType<typeof parseHourlyData>>([]);
  const [funnelData, setFunnelData] = useState<any | null>(null);
  // Account-level (campaign no las tiene en backend) — para que el layout coincida
  const [devices, setDevices] = useState<any[]>([]);
  const [videoData, setVideoData] = useState<any | null>(null);

  // Staged loading
  const [loadingMeta,      setLoadingMeta]      = useState(true);
  const [loadingChart,     setLoadingChart]     = useState(true);
  const [loadingSecondary, setLoadingSecondary] = useState(true);
  const [loadingAudience,  setLoadingAudience]  = useState(true);
  // Backwards-compat flag for existing JSX checks
  const loading = loadingMeta || loadingChart || loadingSecondary || loadingAudience;

  const [leftMetric, setLeftMetric]   = useState<string>("spend");
  const [rightMetric, setRightMetric] = useState<string>("ctr");
  const [stackMetric, setStackMetric] = useState<"spend"|"reach"|"clicks"|"leads">("spend");
  const [costMetric, setCostMetric]   = useState<"cpc"|"cpl">("cpc");
  const [placementToggle, setPlacementToggle] = useState<"spend"|"ctr">("spend");
  const [hourlyMetric, setHourlyMetric] = useState<"spend"|"ctr"|"clicks">("spend");

  const getSinceUntil = () => {
    const today = new Date();
    const until = today.toISOString().split("T")[0];
    switch (period) {
      case "last_7":  { const d = new Date(today); d.setDate(d.getDate()-7);  return { since: d.toISOString().split("T")[0], until }; }
      case "last_30": { const d = new Date(today); d.setDate(d.getDate()-30); return { since: d.toISOString().split("T")[0], until }; }
      case "last_90": { const d = new Date(today); d.setDate(d.getDate()-90); return { since: d.toISOString().split("T")[0], until }; }
      case "this_month":
      default: {
        const since = new Date(today.getFullYear(), today.getMonth(), 1).toISOString().split("T")[0];
        return { since, until };
      }
    }
  };
  const { since, until } = getSinceUntil();

  const cachedDataRef = useRef(cachedData);
  useEffect(() => { cachedDataRef.current = cachedData; }, [cachedData]);

  const onCacheDataRef = useRef(onCacheData);
  useEffect(() => { onCacheDataRef.current = onCacheData; }, [onCacheData]);

  const snapshotRef = useRef<Record<string, any>>({});

  // Sincronizar cuando cambien las props
  useEffect(() => {
    if (initialAds && initialAds.length > 0) setAds(initialAds);
  }, [initialAds]);
  useEffect(() => {
    if (initialAdsets && initialAdsets.length > 0) setAdsets(initialAdsets);
  }, [initialAdsets]);

  const loadData = useCallback(async () => {
    if (!campaign?.campaign_id) return;
    const cid = campaign.campaign_id;

    // Cache hit → restaurar y salir
    const cached = cachedDataRef.current;
    if (cached) {
      setCampaignMeta(cached.campaignMeta ?? null);
      setDailyData(cached.dailyData ?? []);
      setAdsDailyData(cached.adsDailyData ?? []);
      if (cached.ads) setAds(cached.ads);
      if (cached.adsets) setAdsets(cached.adsets);
      setPlacements(cached.placements ?? []);
      setAudienceData(cached.audienceData ?? []);
      setHourlyData(cached.hourlyData ?? []);
      if (cached.funnelData) setFunnelData(cached.funnelData);
      if (cached.devices) setDevices(cached.devices);
      if (cached.videoData) setVideoData(cached.videoData);
      setLoadingMeta(false);
      setLoadingChart(false);
      setLoadingSecondary(false);
      setLoadingAudience(false);
      return;
    }

    setRefreshing(true);
    setLoadingMeta(true);
    setLoadingChart(true);
    setLoadingSecondary(true);
    setLoadingAudience(true);
    snapshotRef.current = {};

    // ── Tanda 1: header + chart ──────────────────────────────────
    Promise.allSettled([
      callAds("campaign_meta",  { campaign_id: cid }),
      callAds("campaign_daily", { campaign_id: cid, since, until }),
    ]).then(([metaRes, dailyRes]) => {
      const freshMeta  = metaRes.status === "fulfilled" ? (metaRes.value as any) : null;
      const freshDaily = dailyRes.status === "fulfilled" ? ((dailyRes.value as any)?.data ?? []) : [];
      if (metaRes.status === "fulfilled") setCampaignMeta(freshMeta);
      if (dailyRes.status === "fulfilled") setDailyData(freshDaily);
      setLoadingMeta(false);
      setLoadingChart(false);

      snapshotRef.current = {
        ...snapshotRef.current,
        campaignMeta: freshMeta,
        dailyData: freshDaily,
      };

      // ── Tanda 2: secundarios ──
      Promise.allSettled([
        callAds("ads",                  { since, until }),
        callAds("adsets",               { since, until }),
        callAds("campaign_ads_daily",   { campaign_id: cid, since, until }),
        callAds("campaign_placement",   { campaign_id: cid, since, until }),
        callAds("conversion_funnel",    { campaign_id: cid, since, until }),
        callAds("device_breakdown",     { campaign_id: cid, since, until }),
        callAds("video_metrics",        { since, until }),
      ]).then(([adsRes, adsetsRes, adsDailyRes, placementRes, funnelRes, devicesRes, videoRes]) => {
        const allAds    = adsRes.status    === "fulfilled" ? ((adsRes.value as any)?.data ?? [])    : [];
        const allAdsets = adsetsRes.status === "fulfilled" ? ((adsetsRes.value as any)?.data ?? []) : [];
        const freshAds    = cid ? allAds.filter((a: any) => !a.campaign_id || a.campaign_id === cid) : allAds;
        const freshAdsets = cid ? allAdsets.filter((a: any) => !a.campaign_id || a.campaign_id === cid) : allAdsets;
        const freshAdsDaily  = adsDailyRes.status  === "fulfilled" ? ((adsDailyRes.value as any)?.data ?? [])  : [];
        const freshPlacement = placementRes.status === "fulfilled" ? ((placementRes.value as any)?.data ?? []) : [];
        const funnelArr      = funnelRes.status    === "fulfilled" ? ((funnelRes.value as any)?.data ?? [])    : [];
        const freshFunnel    = funnelArr[0] ?? null;
        const freshDevices   = devicesRes.status   === "fulfilled" ? ((devicesRes.value as any)?.data ?? [])   : [];
        const videoArr       = videoRes.status     === "fulfilled" ? ((videoRes.value as any)?.data ?? [])     : [];
        const freshVideo     = videoArr[0] ?? null;

        if (adsRes.status === "fulfilled") setAds(freshAds);
        if (adsetsRes.status === "fulfilled") setAdsets(freshAdsets);
        if (adsDailyRes.status === "fulfilled") setAdsDailyData(freshAdsDaily);
        if (placementRes.status === "fulfilled") setPlacements(freshPlacement);
        if (funnelRes.status === "fulfilled") setFunnelData(freshFunnel);
        if (devicesRes.status === "fulfilled") setDevices(freshDevices);
        if (videoRes.status === "fulfilled") setVideoData(freshVideo);
        setLoadingSecondary(false);

        snapshotRef.current = {
          ...snapshotRef.current,
          ads: freshAds,
          adsets: freshAdsets,
          adsDailyData: freshAdsDaily,
          placements: freshPlacement,
          funnelData: freshFunnel,
          devices: freshDevices,
          videoData: freshVideo,
        };
      }).catch(() => setLoadingSecondary(false));

      // ── Tanda 3: audiencia + horario (paralela a tanda 2) ──
      Promise.allSettled([
        callAds("campaign_audience", { campaign_id: cid, since, until }),
        callAds("campaign_hourly",   { campaign_id: cid, since, until }),
      ]).then(([audRes, hrRes]) => {
        const freshAudience = audRes.status === "fulfilled" ? parseAudienceData((audRes.value as any)?.data ?? []) : [];
        const freshHourly   = hrRes.status  === "fulfilled" ? parseHourlyData((hrRes.value as any)?.data ?? [])    : [];
        setAudienceData(freshAudience);
        setHourlyData(freshHourly);
        setLoadingAudience(false);
        setRefreshing(false);

        onCacheDataRef.current?.({
          ...snapshotRef.current,
          audienceData: freshAudience,
          hourlyData: freshHourly,
        });
      }).catch(() => {
        setLoadingAudience(false);
        setRefreshing(false);
      });
    }).catch(() => {
      setLoadingMeta(false);
      setLoadingChart(false);
      setRefreshing(false);
    });
  }, [campaign?.campaign_id, since, until]);

  useEffect(() => { loadData(); }, [loadData]);

  // ── Scroll al tope al abrir el detalle de campaña ─────────────────────────
  useEffect(() => {
    // Buscar el contenedor scrollable más cercano (el shell de admin lo es)
    const scrollables = document.querySelectorAll<HTMLElement>(".admin-shell .overflow-auto");
    scrollables.forEach((el) => el.scrollTo({ top: 0, behavior: "auto" }));
    window.scrollTo({ top: 0, behavior: "auto" });
  }, [campaign?.campaign_id]);

  // ── Tabla de métricas por día (campaign-level) ────────────────────────────
  const campaignDailyRows = dailyData.map((d: any) => ({
    date:        d.date_start ?? "",
    spend:       Number(d.spend ?? 0),
    impressions: Number(d.impressions ?? 0),
    reach:       Number(d.reach ?? 0),
    clicks:      Number(d.clicks ?? 0),
    ctr:         Number(d.ctr ?? 0),
    cpm:         Number(d.cpm ?? 0),
    cpc:         Number(d.cpc ?? 0),
    leads:       getLeads(d.actions ?? []),
    frequency:   Number(d.frequency ?? 0),
  }));
  type CampaignDailyRow = (typeof campaignDailyRows)[number];
  const {
    sorted: sortedCampaignDaily,
    sortKey: campDailySortKey,
    sortDir: campDailySortDir,
    handleSort: campDailySort,
  } = useSortable<CampaignDailyRow>(campaignDailyRows, "date", "desc");
  const campDailyTotals = {
    spend:       sortedCampaignDaily.reduce((s, d) => s + d.spend, 0),
    impressions: sortedCampaignDaily.reduce((s, d) => s + d.impressions, 0),
    reach:       sortedCampaignDaily.reduce((s, d) => s + d.reach, 0),
    clicks:      sortedCampaignDaily.reduce((s, d) => s + d.clicks, 0),
    leads:       sortedCampaignDaily.reduce((s, d) => s + d.leads, 0),
  };

  // ── KPIs de la campaña ────────────────────────────────────────────────────
  const totalSpend       = Number(campaign.spend ?? 0);
  const totalImpressions = Number(campaign.impressions ?? 0);
  const totalReach       = Number(campaign.reach ?? 0);
  const totalClicks      = Number(campaign.clicks ?? 0);
  const avgCtr           = Number(campaign.ctr ?? (totalImpressions > 0 ? (totalClicks/totalImpressions)*100 : 0));
  const totalLeads       = getLeads(campaign.actions);
  const cpl              = totalLeads > 0 ? totalSpend / totalLeads : 0;
  const frequency        = totalReach > 0 ? totalImpressions / totalReach : 0;
  const cpm              = totalImpressions > 0 ? (totalSpend / totalImpressions) * 1000 : 0;
  const cpc              = totalClicks > 0 ? totalSpend / totalClicks : 0;

  // ── Tendencias (primera mitad vs segunda mitad) ────────────────────────────
  const trends = useMemo(() => {
    const empty = {
      spend: null as number | null, impressions: null as number | null, reach: null as number | null,
      frequency: null as number | null, clicks: null as number | null,
      cpm: null as number | null, cpc: null as number | null, leads: null as number | null,
    };
    if (dailyData.length < 4) return empty;
    const sorted = [...dailyData].sort((a, b) => (a.date_start ?? "").localeCompare(b.date_start ?? ""));
    const mid = Math.floor(sorted.length / 2);
    const first = sorted.slice(0, mid);
    const second = sorted.slice(mid);
    const avg = (arr: any[], metric: string) => {
      if (arr.length === 0) return 0;
      const s = arr.reduce((acc, d) => {
        const v = metric === "leads" ? getLeads(d.actions ?? []) : Number(d[metric] ?? 0);
        return acc + v;
      }, 0);
      return s / arr.length;
    };
    const trend = (metric: string): number | null => {
      const a = avg(first, metric); const b = avg(second, metric);
      if (a === 0) return null;
      return ((b - a) / a) * 100;
    };
    return {
      spend: trend("spend"), impressions: trend("impressions"), reach: trend("reach"),
      frequency: null as number | null,
      clicks: trend("clicks"), cpm: trend("cpm"), cpc: trend("cpc"), leads: trend("leads"),
    };
  }, [dailyData]);

  // ── Combo chart data (campaign daily) ────────────────────────────────────
  const accountChartData = useMemo(() => [...dailyData]
    .sort((a, b) => (a.date_start ?? "").localeCompare(b.date_start ?? ""))
    .map((d: any) => ({
      date:        d.date_start?.slice(5),
      spend:       Number(d.spend ?? 0),
      impressions: Number(d.impressions ?? 0),
      clicks:      Number(d.clicks ?? 0),
      ctr:         Number(d.ctr ?? 0),
      reach:       Number(d.reach ?? 0),
      leads:       getLeads(d.actions ?? []),
      cpm:         Number(d.cpm ?? 0),
      cpc:         Number(d.cpc ?? 0),
    })), [dailyData]);

  // ── Comparativa por anuncio (una barra por ad, color sólido) ─────────────
  const adComparison = useMemo(() => {
    return ads.map((ad: any) => {
      const name = (ad.ad_name ?? ad.ad_id ?? "ad").toString();
      const val = stackMetric === "leads"
        ? getLeads(ad.actions ?? [])
        : stackMetric === "clicks"
        ? Number(ad.inline_link_clicks ?? ad.clicks ?? 0)
        : Number(ad[stackMetric] ?? 0);
      return { name, value: val };
    }).filter(a => a.value > 0)
      .sort((a, b) => b.value - a.value);
  }, [ads, stackMetric]);

  // Mapa estable de color por ad (compartido entre Comparativa y Costo por anuncio)
  const adColorMap = useMemo(() => {
    const map = new Map<string, string>();
    const ordered = ads
      .map((a: any) => (a.ad_name ?? a.ad_id ?? "ad").toString())
      .filter((n, i, arr) => arr.indexOf(n) === i)
      .sort();
    ordered.forEach((name, i) => map.set(name, AD_COLORS[i % AD_COLORS.length]));
    return map;
  }, [ads]);

  // ── Costo por anuncio (line chart CPC/CPL por ad activo) ─────────────────
  const costByAd = useMemo(() => {
    const dateSet = new Set<string>();
    const byAd = new Map<string, { name: string; daily: Map<string, { spend: number; clicks: number; leads: number }> }>();

    for (const row of adsDailyData as any[]) {
      const aid = row.ad_id ?? "unknown";
      const aname = (row.ad_name ?? aid).toString();
      const date = (row.date_start ?? "").slice(5);
      if (!date) continue;
      dateSet.add(date);
      if (!byAd.has(aid)) byAd.set(aid, { name: aname, daily: new Map() });
      const entry = byAd.get(aid)!;
      const prev = entry.daily.get(date) ?? { spend: 0, clicks: 0, leads: 0 };
      entry.daily.set(date, {
        spend:  prev.spend  + Number(row.spend ?? 0),
        clicks: prev.clicks + Number(row.inline_link_clicks ?? row.clicks ?? 0),
        leads:  prev.leads  + getLeads(row.actions ?? []),
      });
    }

    const dates = Array.from(dateSet).sort();
    const activeAds = Array.from(byAd.entries()).filter(([_, a]) =>
      Array.from(a.daily.values()).some(d => d.spend > 0)
    );

    const data = dates.map(date => {
      const entry: Record<string, any> = { date };
      for (const [_aid, a] of activeAds) {
        const d = a.daily.get(date);
        if (!d) continue;
        if (costMetric === "cpc") {
          if (d.clicks > 0) entry[a.name] = Number((d.spend / d.clicks).toFixed(2));
        } else {
          if (d.leads > 0) entry[a.name] = Number((d.spend / d.leads).toFixed(2));
        }
      }
      return entry;
    });

    return { data, ads: activeAds.map(([_, a]) => a.name) };
  }, [adsDailyData, costMetric]);

  // ── Acciones agregadas (de daily) ─────────────────────────────────────────
  const allActionsMap: Record<string, number> = {};

                <div className="h-[140px] flex items-center justify-center text-xs text-muted-foreground">Sin datos</div>
              ) : (
                <>
                  <div className="space-y-1.5">
                    {deviceAgg.map((d, i) => {
                      const pct = deviceTotal > 0 ? (d.spend / deviceTotal) * 100 : 0;
                      return (
                        <div key={i} className="flex items-center gap-2 text-xs">
                          <div className="flex-1 min-w-0 truncate text-muted-foreground">{d.name}</div>
                          <div className="w-12 h-2 bg-muted rounded overflow-hidden">
                            <div className="h-full bg-primary rounded" style={{ width: `${Math.min(pct, 100)}%` }} />
                          </div>
                          <div className="w-14 text-right text-foreground font-medium">{fmtMoney(d.spend)}</div>
                          <div className="w-9 text-right text-[10px] text-muted-foreground tabular-nums">{pct.toFixed(0)}%</div>
                        </div>
                      );
                    })}
                  </div>
                  <div className="pt-2 border-t border-border space-y-1">
                    <p className="text-[10px] uppercase tracking-wider text-muted-foreground">CTR</p>
                    {deviceAgg.map((d, i) => {
                      const ctrClass = d.ctr > 3.5 ? "text-green-600" : d.ctr < 2 ? "text-destructive" : "text-muted-foreground";
                      return (
                        <div key={i} className="flex items-center justify-between text-xs">
                          <span className="text-foreground truncate">{d.name}</span>
                          <span className={ctrClass}>{d.ctr.toFixed(2)}%</span>
                        </div>
                      );
                    })}
                  </div>
                </>
              )}
            </div>

            {/* Retención de video */}
            <div className="rounded-xl border border-border bg-card p-3 space-y-3">
              <h3 className="text-sm font-semibold text-foreground flex items-center gap-2">
                <Zap className="w-4 h-4 text-primary" /> Retención de video
              </h3>
              {loading && !videoData ? (
                <div className="h-[140px] flex items-center justify-center"><Loader2 className="w-5 h-5 animate-spin text-primary" /></div>
              ) : !videoStages ? (
                <div className="h-[140px] flex flex-col items-center justify-center text-xs text-muted-foreground gap-2">
                  <VideoOff className="w-7 h-7 opacity-40" /> Sin datos de video
                </div>
              ) : (
                <>
                  <div className="flex items-end justify-between gap-1 h-[95px] pt-5">
                    {videoStages.map((s, i) => (
                      <div key={i} className="flex-1 flex flex-col items-center gap-1">
                        <span className="text-[9px] font-bold text-foreground">{s.pct.toFixed(0)}%</span>
                        <div className="w-full bg-primary rounded-t" style={{ height: `${Math.max((s.pct / 100) * 75, 2)}px` }} />
                      </div>
                    ))}
                  </div>
                  <div className="flex justify-between gap-1">
                    {videoStages.map((s, i) => (
                      <span key={i} className="flex-1 text-center text-[9px] text-muted-foreground">{s.label}</span>
                    ))}
                  </div>
                  <div className="pt-2 border-t border-border flex justify-between text-[11px]">
                    <span className="text-muted-foreground">T. prom <span className="text-foreground font-semibold">{avgWatchTime.toFixed(1)}s</span></span>
                    <span className="text-muted-foreground">Thru <span className="text-foreground font-semibold">{fmt(thruPlays)}</span></span>
                  </div>
                </>
              )}
            </div>
          </div>

          {/* SECCIÓN — AUDIENCIA full width */}
          {audienceData.length > 0 && (
            <div className="rounded-xl border border-border bg-card p-3 space-y-3">
              <h3 className="text-sm font-semibold text-foreground flex items-center gap-2">
                <Users className="w-4 h-4 text-primary" /> Audiencia — Edad & Género
              </h3>
              <AudienceInsights data={audienceData} />
            </div>
          )}

          {/* SECCIÓN — Desglose horario */}
          {hourlyData.some((h) => h.spend > 0 || h.clicks > 0) && (
            <div className="rounded-xl border border-border bg-card p-3 space-y-3">
              <div className="flex items-center justify-between flex-wrap gap-2">
                <h3 className="text-sm font-semibold text-foreground flex items-center gap-2">
                  <Clock className="w-4 h-4 text-primary" /> Rendimiento por hora del día
                </h3>
                <div className="flex gap-1">
                  {(["spend","ctr","clicks"] as const).map((m) => (
                    <button key={m} onClick={() => setHourlyMetric(m)}
                      className={`px-2.5 py-1 rounded-lg text-xs transition-colors ${
                        hourlyMetric === m ? "bg-primary text-primary-foreground" : "bg-muted text-muted-foreground hover:text-foreground"
                      }`}>
                      {m === "spend" ? "Gasto" : m === "ctr" ? "CTR" : "Clics"}
                    </button>
                  ))}
                </div>
              </div>
              <HourlyChart data={hourlyData} metric={hourlyMetric} />
            </div>
          )}

          {/* SECCIÓN — Funnel de conversión */}
          {funnelData && (() => {
            const fImpr = Number(funnelData.impressions ?? 0);
            const fClicks = Number(funnelData.outbound_clicks?.[0]?.value ?? funnelData.clicks ?? 0);
            const fLeads = getLeadsFromActions(funnelData.actions ?? []);
            const fPurchases = getPurchases(funnelData.actions ?? []);
            const fRoas = getRoas(funnelData.purchase_roas ?? []);
            const fConvValue = getConversionValue(funnelData.action_values ?? []);
            if (fLeads === 0 && fPurchases === 0) return null;
            const steps = [
              { label: "Impresiones", value: fImpr, pct: 100, fmt: fImpr >= 1000 ? `${(fImpr/1000).toFixed(1)}k` : String(fImpr) },
              { label: "Clics", value: fClicks, pct: fImpr > 0 ? (fClicks/fImpr)*100 : 0, fmt: String(fClicks) },
              { label: "Leads", value: fLeads, pct: fClicks > 0 ? (fLeads/fClicks)*100 : 0, fmt: String(fLeads) },
              ...(fPurchases > 0 ? [{
                label: "Compras", value: fPurchases,
                pct: fLeads > 0 ? (fPurchases/fLeads)*100 : (fClicks > 0 ? (fPurchases/fClicks)*100 : 0),
                fmt: String(fPurchases),
              }] : []),
            ];
            return (
              <div className="rounded-xl border border-border bg-card p-3 space-y-3">
                <h3 className="text-sm font-semibold text-foreground flex items-center gap-2">
                  <TrendingDown className="w-4 h-4 text-primary" /> Funnel de conversión
                </h3>
                <FunnelViz steps={steps} roas={fRoas} convValue={fConvValue} />
              </div>
            );
          })()}

          {/* SECCIÓN — DESGLOSE JERÁRQUICO (AdSets / Anuncios) */}
          <div className="space-y-2">
            <p className="text-xs uppercase tracking-wider text-muted-foreground font-medium">Desglose jerárquico</p>
            <div className="rounded-xl border border-border bg-card p-4">
              <Tabs defaultValue="adsets">
                <TabsList className="mb-3">
                  <TabsTrigger value="adsets">Ad Sets</TabsTrigger>
                  <TabsTrigger value="ads">Anuncios</TabsTrigger>
                </TabsList>

                <TabsContent value="adsets">
                  {loading && adsets.length === 0 ? (
                    <div className="flex items-center justify-center py-12"><Loader2 className="w-5 h-5 animate-spin text-primary" /></div>
                  ) : adsets.length === 0 ? (
                    <div className="flex flex-col items-center justify-center py-10 text-xs text-muted-foreground gap-2">
                      <BarChart2 className="w-8 h-8 opacity-40" /> Sin ad sets en este período
                    </div>
                  ) : (
                    <div className="overflow-x-auto">
                      <table className="w-full text-sm">
                        <thead>
                          <tr className="border-b border-border">
                            {["Ad Set","Estado","Gasto","Impr.","Alcance","Frec.","Clics","CTR","CPM","CPC","Leads","CPL"].map((h, i) => (
                              <th key={h} className={`py-2 px-2 text-xs uppercase tracking-wider font-medium text-muted-foreground ${i >= 2 ? "text-right" : "text-left"}`}>{h}</th>
                            ))}
                          </tr>
                        </thead>
                        <tbody>
                          {adsets.map((a: any, i) => {
                            const aImpr = Number(a.impressions ?? 0);
                            const aReach = Number(a.reach ?? 0);
                            const aClicks = Number(a.clicks ?? 0);
                            const aSpend = Number(a.spend ?? 0);
                            const aCtr = Number(a.ctr ?? 0);
                            const aFreq = aReach > 0 ? aImpr / aReach : 0;
                            const aCpm = Number(a.cpm ?? (aImpr > 0 ? (aSpend / aImpr) * 1000 : 0));
                            const aCpc = aClicks > 0 ? aSpend / aClicks : 0;
                            const aLeads = getLeads(a.actions);
                            const aCpl = aLeads > 0 ? aSpend / aLeads : 0;
                            const ctrClass = aCtr > 3.5 ? "text-green-600" : aCtr < 2 && aCtr > 0 ? "text-destructive" : "text-foreground";
                            return (
                              <tr key={i} className="border-b border-border/50 hover:bg-muted/30">
                                <td className="py-2 px-2 max-w-[200px] truncate font-medium text-foreground" title={a.adset_name}>{a.adset_name}</td>
                                <td className="py-2 px-2 text-xs"><StatusDot status={a.status} /></td>
                                <td className="py-2 px-2 text-right text-foreground">{fmtMoney(aSpend)}</td>
                                <td className="py-2 px-2 text-right text-foreground">{fmt(aImpr)}</td>
                                <td className="py-2 px-2 text-right text-foreground">{fmt(aReach)}</td>
                                <td className="py-2 px-2 text-right text-foreground">{aFreq > 0 ? aFreq.toFixed(2) : "—"}</td>
                                <td className="py-2 px-2 text-right text-foreground">{fmt(aClicks)}</td>
                                <td className={`py-2 px-2 text-right ${ctrClass}`}>{aCtr.toFixed(2)}%</td>
                                <td className="py-2 px-2 text-right text-foreground">{fmtMoney(aCpm)}</td>
                                <td className="py-2 px-2 text-right text-foreground">{fmtMoney(aCpc)}</td>
                                <td className={`py-2 px-2 text-right ${aLeads === 0 ? "text-destructive font-bold" : "text-foreground font-semibold"}`}>{fmt(aLeads)}</td>
                                <td className="py-2 px-2 text-right text-foreground">{aLeads > 0 ? fmtMoney(aCpl) : "—"}</td>
                              </tr>
                            );
                          })}
                        </tbody>
                      </table>
                    </div>
                  )}
                </TabsContent>

                <TabsContent value="ads">
                  {loading && ads.length === 0 ? (
                    <div className="flex items-center justify-center py-12"><Loader2 className="w-5 h-5 animate-spin text-primary" /></div>
                  ) : ads.length === 0 ? (
                    <div className="flex flex-col items-center justify-center py-10 text-xs text-muted-foreground gap-2">
                      <BarChart2 className="w-8 h-8 opacity-40" /> Sin anuncios en este período
                    </div>
                  ) : (
                    <div className="overflow-x-auto">
                      <table className="w-full text-sm">
                        <thead>
                          <tr className="border-b border-border">
                            {["Anuncio","Gasto","Impr.","Clics","CTR","CPC","Leads","Calidad","Eng.","Conv."].map((h, i) => (
                              <th key={h} className={`py-2 px-2 text-xs uppercase tracking-wider font-medium text-muted-foreground ${i === 0 ? "text-left" : "text-right"}`}>{h}</th>
                            ))}
                          </tr>
                        </thead>
                        <tbody>
                          {ads.map((a: any, i: number) => {
                            const aSpend = Number(a.spend ?? 0);
                            const aImpr = Number(a.impressions ?? 0);
                            const aClicks = Number(a.inline_link_clicks ?? a.clicks ?? 0);
                            const aCtr = Number(a.ctr ?? 0);
                            const aCpc = Number(a.cpc ?? (aClicks > 0 ? aSpend / aClicks : 0));
                            const aLeads = getLeads(a.actions);
                            const ctrClass = aCtr > 3.5 ? "text-green-600" : aCtr < 2 && aCtr > 0 ? "text-destructive" : "text-foreground";
                            return (
                              <tr key={i}
                                onClick={() => onOpenAd?.(a)}
                                className={`border-b border-border/50 ${onOpenAd ? "cursor-pointer hover:bg-muted/50" : "hover:bg-muted/30"}`}>
                                <td className="py-2 px-2 max-w-[220px] truncate font-medium text-foreground" title={a.ad_name}>{a.ad_name}</td>
                                <td className="py-2 px-2 text-right text-foreground">{fmtMoney(aSpend)}</td>
                                <td className="py-2 px-2 text-right text-foreground">{fmt(aImpr)}</td>
                                <td className="py-2 px-2 text-right text-foreground">{fmt(aClicks)}</td>
                                <td className={`py-2 px-2 text-right ${ctrClass}`}>{aCtr.toFixed(2)}%</td>
                                <td className="py-2 px-2 text-right text-foreground">{fmtMoney(aCpc)}</td>
                                <td className={`py-2 px-2 text-right ${aLeads === 0 ? "text-destructive font-bold" : "text-foreground font-semibold"}`}>{fmt(aLeads)}</td>
                                <td className="py-2 px-2 text-right"><RankBadge value={a.quality_ranking} /></td>
                                <td className="py-2 px-2 text-right"><RankBadge value={a.engagement_rate_ranking} /></td>
                                <td className="py-2 px-2 text-right"><RankBadge value={a.conversion_rate_ranking} /></td>
                              </tr>
                            );
                          })}
                        </tbody>
                      </table>
                    </div>
                  )}
                </TabsContent>
              </Tabs>
            </div>
          </div>

          {/* SECCIÓN — MÉTRICAS POR DÍA (campaign-level) */}
          {sortedCampaignDaily.length > 0 && (
            <div className="rounded-xl border border-border bg-card p-3 space-y-3">
              <h3 className="text-sm font-semibold text-foreground flex items-center gap-2">
                <CalendarDays className="w-4 h-4 text-primary" /> Métricas por día
              </h3>
              <div className="overflow-x-auto">
                <table className="w-full text-sm">
                  <thead>
                    <tr className="border-b border-border">
                      <SortableTh<CampaignDailyRow> label="Fecha"   sortKey="date"        activeKey={campDailySortKey} dir={campDailySortDir} onSort={campDailySort} right={false} />
                      <SortableTh<CampaignDailyRow> label="Gasto"   sortKey="spend"       activeKey={campDailySortKey} dir={campDailySortDir} onSort={campDailySort} />
                      <SortableTh<CampaignDailyRow> label="Impr."   sortKey="impressions" activeKey={campDailySortKey} dir={campDailySortDir} onSort={campDailySort} />
                      <SortableTh<CampaignDailyRow> label="Alcance" sortKey="reach"       activeKey={campDailySortKey} dir={campDailySortDir} onSort={campDailySort} />
                      <SortableTh<CampaignDailyRow> label="Frec."   sortKey="frequency"   activeKey={campDailySortKey} dir={campDailySortDir} onSort={campDailySort} />
                      <SortableTh<CampaignDailyRow> label="Clics"   sortKey="clicks"      activeKey={campDailySortKey} dir={campDailySortDir} onSort={campDailySort} />
                      <SortableTh<CampaignDailyRow> label="CTR"     sortKey="ctr"         activeKey={campDailySortKey} dir={campDailySortDir} onSort={campDailySort} />
                      <SortableTh<CampaignDailyRow> label="CPM"     sortKey="cpm"         activeKey={campDailySortKey} dir={campDailySortDir} onSort={campDailySort} />
                      <SortableTh<CampaignDailyRow> label="CPC"     sortKey="cpc"         activeKey={campDailySortKey} dir={campDailySortDir} onSort={campDailySort} />
                      <SortableTh<CampaignDailyRow> label="Leads"   sortKey="leads"       activeKey={campDailySortKey} dir={campDailySortDir} onSort={campDailySort} />
                    </tr>
                  </thead>
                  <tbody>
                    {sortedCampaignDaily.map((d, i) => (
                      <tr key={i} className="border-b border-border/50 hover:bg-muted/30">
                        <td className="py-2 px-2 text-foreground">{d.date}</td>
                        <td className="py-2 px-2 text-right text-foreground">${d.spend.toFixed(2)}</td>
                        <td className="py-2 px-2 text-right text-foreground">{d.impressions.toLocaleString()}</td>
                        <td className="py-2 px-2 text-right text-foreground">{d.reach.toLocaleString()}</td>
                        <td className="py-2 px-2 text-right text-foreground">{d.frequency > 0 ? d.frequency.toFixed(2) : "—"}</td>
                        <td className="py-2 px-2 text-right text-foreground">{d.clicks.toLocaleString()}</td>
                        <td className="py-2 px-2 text-right text-foreground">{d.ctr.toFixed(2)}%</td>
                        <td className="py-2 px-2 text-right text-foreground">${d.cpm.toFixed(2)}</td>
                        <td className="py-2 px-2 text-right text-foreground">${d.cpc.toFixed(2)}</td>
                        <td className={`py-2 px-2 text-right ${d.leads === 0 ? "text-muted-foreground" : "text-foreground font-semibold"}`}>{d.leads}</td>
                      </tr>
                    ))}
                  </tbody>
                  <tfoot>
                    <tr className="border-t border-border font-semibold">
                      <td className="py-2 px-2 text-foreground">Total</td>
                      <td className="py-2 px-2 text-right text-foreground">${campDailyTotals.spend.toFixed(2)}</td>
                      <td className="py-2 px-2 text-right text-foreground">{campDailyTotals.impressions.toLocaleString()}</td>
                      <td className="py-2 px-2 text-right text-foreground">{campDailyTotals.reach.toLocaleString()}</td>
                      <td className="py-2 px-2 text-right text-muted-foreground">—</td>
                      <td className="py-2 px-2 text-right text-foreground">{campDailyTotals.clicks.toLocaleString()}</td>
                      <td className="py-2 px-2 text-right text-muted-foreground">—</td>
                      <td className="py-2 px-2 text-right text-muted-foreground">—</td>
                      <td className="py-2 px-2 text-right text-muted-foreground">—</td>
                      <td className="py-2 px-2 text-right text-foreground">{campDailyTotals.leads}</td>
                    </tr>
                  </tfoot>
                </table>
              </div>
            </div>
          )}
        </div>
      </div>
    </TooltipProvider>
  );
}
